\textbf{Phần 1. Trắc nghiệm lựa chọn}
\setcounterpageref{ex}{0}

\begin{ex} %[4D5N24]
Tính chất giao hoán của phép cộng được viết dưới dạng công thức là
\choice
{$a + b = a - b$}
{$(a + b) + c = a + (b + c)$}
{\True $a + b = b + a$}
{$a + 0 = a$}
\loigiai{
$a + b = b + a$.\\
\textbf{Đáp án của câu hỏi là C}
}
\end{ex}

\begin{ex} %[4D5N24]
Tính chất kết hợp của phép cộng được viết dưới dạng công thức là
\choice
{$a + b = b + a$}
{\True $(a + b) + c = a + (b + c)$}
{$a \times b = b \times a$}
{$a + (b - c) = a + b - c$}
\loigiai{
$(a + b) + c = a + (b + c)$.\\
\textbf{Đáp án của câu hỏi là B}
}
\end{ex}

\begin{ex} %[4D5H24]
Điền số thích hợp vào chỗ chấm: $45\,678 + 12\,345 = 12\,345 + \dots$
\choice
{$12\,345$}
{$58\,023$}
{\True $45\,678$}
{$45\,000$}
\loigiai{
Theo tính chất giao hoán: $45\,678 + 12\,345 = 12\,345 + 45\,678$.\\
\textbf{Đáp án của câu hỏi là C}
}
\end{ex}

\textbf{Phần 2. Trắc nghiệm đúng sai}
\setcounterpageref{ex}{0}

\begin{ex} %[4D5H24]
Xét tính đúng sai của các biểu thức áp dụng tính chất phép cộng:
\choiceTF
{\True $a + b + c = (a + c) + b$.}
{$(a + b) + c = a + (b - c)$.}
{\True $125 + 456 + 875 = (125 + 875) + 456 = 1456$.}
{\True $(a + 0) = (0 + a) = a$.}
\loigiai{
\begin{itemize}
    \item Giao hoán và kết hợp: Đúng.
    \item Phép cộng không biến thành phép trừ. Sai.
    \item Tính thuận tiện đúng.
    \item Cộng với 0: Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Sai}.
    \item c) \textbf{Đúng}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\begin{ex} %[4D5V24]
Tính giá trị các biểu thức:
$A = 250 + 480 + 750$ và $B = 360 + 640 + 480$.
Xét tính đúng sai:
\choiceTF
{\True $A = 1480$.}
{\True $B = 1480$.}
{Giá trị biểu thức $A$ lớn hơn giá trị biểu thức $B$.}
{\True $A = B$.}
\loigiai{
\begin{itemize}
    \item $A = (250 + 750) + 480 = 1000 + 480 = 1480$. Đúng.
    \item $B = (360 + 640) + 480 = 1000 + 480 = 1480$. Đúng.
    \item $A = B$ nên khẳng định $A > B$ là Sai.
    \item $A = B$. Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\textbf{Phần 3. Trả lời ngắn}
\setcounterpageref{ex}{0}

\begin{ex} %[4D5H24]
Tính thuận tiện: $340 + 256 + 660$.
\shortans{1256}
\loigiai{
\textbf{Đáp số:} $1256$\\
$(340 + 660) + 256 = 1256$.
}
\end{ex}

\begin{ex} %[4D5H24]
Tính thuận tiện: $1500 + 2800 + 8500$.
\shortans{12800}
\loigiai{
\textbf{Đáp số:} $12800$\\
$(1500 + 8500) + 2800 = 12\,800$.
}
\end{ex}

\begin{ex} %[4D5V24]
Tính nhanh: $45 + 189 + 55 + 211$.
\shortans{500}
\loigiai{
\textbf{Đáp số:} $500$\\
$(45 + 55) + (189 + 211) = 100 + 400 = 500$.
}
\end{ex}

\begin{ex} %[4D5V24]
Tính nhanh: $12\,000 + 34\,000 + 88\,000 + 66\,000$.
\shortans{200000}
\loigiai{
\textbf{Đáp số:} $200000$\\
$(12\,000 + 88\,000) + (34\,000 + 66\,000) = 100\,000 + 100\,000 = 200\,000$.
}
\end{ex}

\begin{ex} %[4D5V24]
Tính tổng của 20 số tự nhiên liên tiếp từ 1 đến 20.
\shortans{210}
\loigiai{
\textbf{Đáp số:} $210$\\
$(1 + 20) \times 20 : 2 = 210$.
}
\end{ex}

\begin{ex} %[4D5C24]
Tính tổng: $11 + 22 + 33 + 44 + 55 + 66 + 77 + 88 + 99$.
\shortans{495}
\loigiai{
\textbf{Đáp số:} $495$\\
$11 \times (1 + 2 + \dots + 9) = 11 \times 45 = 495$.
}
\end{ex}
