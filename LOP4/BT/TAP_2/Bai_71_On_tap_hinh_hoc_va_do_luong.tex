\begin{ex}
Hình bình hành có đáy 20 cm, chiều cao 8 cm. Diện tích là bao nhiêu xăng-ti-mét vuông?
\shortans{160}
\loigiai{
$20 \times 8 = 160\text{ cm}^2$.
}
\end{ex}

\begin{ex}
Hình thoi có hai đường chéo dài 15 cm và 8 cm. Diện tích là bao nhiêu xăng-ti-mét vuông?
\shortans{60}
\loigiai{
$15 \times 8 : 2 = 60\text{ cm}^2$.
}
\end{ex}

\begin{ex}

\loigiai{

}
\end{ex}

\begin{ex}

\end{ex}
