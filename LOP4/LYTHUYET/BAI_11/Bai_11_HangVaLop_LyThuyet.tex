    arc=2.5mm, boxrule=0.8pt, top=1.8mm, bottom=1.8mm, left=2mm, right=2mm

    \footnotesize
    $\bullet$ \textbf{Tách lớp:} Tách số thành từng nhóm 3 chữ số từ phải sang trái.\\
    $\bullet$ \textbf{Đọc số:} Đọc lần lượt từ trái sang phải, đọc như đọc số có 3 chữ số ở mỗi lớp rồi \textbf{thêm tên của lớp đó} (trừ lớp đơn vị không cần đọc tên lớp).\\
    \textit{Ví dụ:} $452\,318$: Bốn trăm năm mươi hai \textbf{nghìn} ba trăm mười tám.
  \end{tcolorbox}

\begin{minipage}[t]{0.54\textwidth}
  \begin{tcolorbox}[
    enhanced,
    title={\footnotesize\bfseries\color{white}\faKey\ Phân tích số theo từng lớp},
    colback=green!3!white,
    colframe=green!60!black,
    arc=2.5mm, boxrule=0.8pt, top=1.8mm, bottom=1.8mm, left=2.2mm, right=2.2mm
  ]
    \footnotesize
    $\bullet$ \textbf{Ví dụ số:} $345\,678$\\
    - Thuộc \textbf{lớp nghìn} gồm các chữ số: $3, 4, 5$.\\
    - Thuộc \textbf{lớp đơn vị} gồm các chữ số: $6, 7, 8$.\\
    $\bullet$ Phân tích: $345\,678 = 345\,000 + 678$.\\
    $\bullet$ Giá trị của chữ số $3$ là $300\,000$ (hàng trăm nghìn).
  \end{tcolorbox}
\end{minipage}%
