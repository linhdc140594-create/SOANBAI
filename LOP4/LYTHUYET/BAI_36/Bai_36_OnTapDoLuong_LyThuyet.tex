    $\bullet$ \textbf{Khối lượng (gấp kém nhau 10 lần)}:
    \begin{center}
      \textbf{Tấn $\rightarrow$ Tạ $\rightarrow$ Yến $\rightarrow$ kg $\rightarrow$ g}\\
      $1\text{ tấn} = 10\text{ tạ} = 100\text{ yến} = 1\,000\text{ kg}$\\
      $1\text{ yến} = 10\text{ kg}$; $1\text{ kg} = 1\,000\text{ g}$.
    \end{center}
    $\bullet$ \textbf{Diện tích (gấp kém nhau 100 lần)}:
    \begin{center}
      \textbf{$\text{m}^2 \rightarrow \text{dm}^2 \rightarrow \text{cm}^2 \rightarrow \text{mm}^2$}\\
      $1\text{ m}^2 = 100\text{ dm}^2 = 10\,000\text{ cm}^2 = 1\,000\,000\text{ mm}^2$
    \end{center}
    Mỗi đơn vị diện tích ứng với đúng \textbf{2 chữ số}.
  \end{tcolorbox}

    $\bullet$ \textbf{Các đơn vị thời gian cốt lõi}:
    \begin{center}
      $1\text{ thế kỉ} = 100\text{ năm}$\\
      $1\text{ năm} = 12\text{ tháng}$; $1\text{ tuần} = 7\text{ ngày}$\\
      $1\text{ ngày} = 24\text{ giờ}$; $1\text{ giờ} = 60\text{ phút}$\\
      $1\text{ phút} = 60\text{ giây}$; $1\text{ giờ} = 3\,600\text{ giây}$.
    \end{center}
    $\bullet$ \textbf{Quy tắc xác định thế kỉ}:\\
    Năm $abcd$: Nếu $cd = 00 \Rightarrow$ thế kỉ $ab$. Nếu $cd \ne 00 \Rightarrow$ thế kỉ $ab + 1$.
