    \footnotesize
    $\bullet$ \textbf{Ví dụ 1 (So sánh hai đại lượng):}\\
    So sánh $2\text{ tấn } 50\text{ kg}$ và $2\,500\text{ kg}$:\\
    Đổi $2\text{ tấn } 50\text{ kg} = 2\,050\text{ kg}$.\\
    Vì $2\,050\text{ kg} < 2\,500\text{ kg}$ nên $\mathbf{2\text{ tấn } 50\text{ kg} < 2\,500\text{ kg}}$.\\
    $\bullet$ \textbf{Ví dụ 2 (Tính toán):}\\
    $4\text{ m}^2\ 20\text{ dm}^2 + 3\text{ m}^2\ 85\text{ dm}^2 = 7\text{ m}^2\ 105\text{ dm}^2 = \mathbf{8\text{ m}^2\ 5\text{ dm}^2}$.
  \end{tcolorbox}
