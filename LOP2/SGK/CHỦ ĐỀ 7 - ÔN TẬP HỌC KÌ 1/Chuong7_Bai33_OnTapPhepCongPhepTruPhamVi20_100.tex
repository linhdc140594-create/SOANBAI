\textbf{Phần 2. Trắc nghiệm đúng sai}
\setcounterpageref{ex}{0}

\begin{ex} %[2D7H33]
Xét tính đúng sai của các phép tính:
\choiceTF
{\True $67 + 28 = 95$.}
{\True $92 - 48 = 44$.}
{$15 - 8 = 6$.}
{\True $15 - 8 = 7$.}
\loigiai{
\begin{itemize}
    \item $67 + 28 = 95$: Đúng.
    \item $92 - 48 = 44$: Đúng.
    \item $15 - 8 = 7 \neq 6$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\begin{ex} %[2D7V33]
Bác nông dân có 64 con vịt, bác bán đi 28 con vịt. Xét tính đúng sai:
\choiceTF
{\True Bác còn lại 36 con vịt ($64 - 28 = 36$).}
{\True Số vịt còn lại nhiều hơn số vịt đã bán là 8 con ($36 - 28 = 8$).}
{Nếu bác mua thêm 10 con thì có 40 con vịt.}
{\True Khi đó bác có $36 + 10 = 46$ con vịt.}
\loigiai{
\begin{itemize}
    \item Còn lại: $64 - 28 = 36$: Đúng.
    \item Còn lại hơn đã bán: $36 - 28 = 8$: Đúng.
    \item $36 + 10 = 46 \neq 40$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\textbf{Phần 3. Trả lời ngắn}

Tính: $56 + 37$.
\shortans{93}
\loigiai{
\textbf{Đáp số:} $93$\\
$56 + 37 = 93$.
}
\end{ex}

\begin{ex} %[2D7H33]

\begin{ex} %[2D7V33]
