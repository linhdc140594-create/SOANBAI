\textbf{Phần 2. Trắc nghiệm đúng sai}
\setcounterpageref{ex}{0}

\begin{ex} %[2D2H07]
Xét tính đúng sai của các phép cộng:
\choiceTF
{\True $9 + 6 = 15$.}
{\True $8 + 7 = 15$.}
{$7 + 6 = 14$.}
{\True $7 + 6 = 13$.}
\loigiai{
\begin{itemize}
    \item $9 + 6 = 15$: Đúng.
    \item $8 + 7 = 15$: Đúng.
    \item $7 + 6 = 13 \neq 14$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\begin{ex} %[2D2V07]
Cho biểu thức $A = 9 + 7$ và $B = 8 + 8$. Xét tính đúng sai:
\choiceTF
{\True Giá trị của biểu thức $A$ là 16.}
{\True Giá trị của biểu thức $B$ là 16.}
{Biểu thức $A$ có giá trị lớn hơn biểu thức $B$.}
{\True Hai biểu thức $A$ và $B$ có giá trị bằng nhau ($A = B = 16$).}
\loigiai{
\begin{itemize}
    \item $A = 9 + 7 = 16$: Đúng.
    \item $B = 8 + 8 = 16$: Đúng.
    \item $A = B$ (không lớn hơn). Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}
