\textbf{Phần 2. Trắc nghiệm đúng sai}
\setcounterpageref{ex}{0}

\begin{ex} %[2D4H24]
Xét tính đúng sai của các phép toán:
\choiceTF
{\True $48 + 36 = 84$.}
{\True $84 - 48 = 36$.}
{$72 - 38 = 35$.}
{\True $72 - 38 = 34$.}
\loigiai{
\begin{itemize}
    \item $48 + 36 = 84$: Đúng.
    \item $84 - 48 = 36$: Đúng.
    \item $72 - 38 = 34 \neq 35$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\begin{ex} %[2D4V24]
Mai có 52 cái kẹo, Mai ăn hết 16 cái kẹo. Xét tính đúng sai:
\choiceTF
{\True Mai còn lại 36 cái kẹo ($52 - 16 = 36$).}
{\True Số kẹo ban đầu nhiều hơn số kẹo còn lại 16 cái.}
{Nếu Mai được mẹ cho thêm 10 cái thì Mai có 45 cái.}
{\True Khi đó Mai có $36 + 10 = 46$ cái kẹo.}
\loigiai{
\begin{itemize}
    \item Còn lại: $52 - 16 = 36$: Đúng.
    \item Ban đầu hơn còn lại 16: Đúng.
    \item $36 + 10 = 46 \neq 45$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\textbf{Phần 3. Trả lời ngắn}
\setcounterpageref{ex}{0}

\begin{ex} %[2D4H24]
Tính: $57 + 38$.
\shortans{95}
\loigiai{
\textbf{Đáp số:} $95$\\
$57 + 38 = 95$.
}
\end{ex}

\begin{ex} %[2D4H24]
