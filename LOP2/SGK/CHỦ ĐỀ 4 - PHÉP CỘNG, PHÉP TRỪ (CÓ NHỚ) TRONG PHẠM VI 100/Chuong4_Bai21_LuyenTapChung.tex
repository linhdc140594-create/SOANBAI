\textbf{Phần 2. Trắc nghiệm đúng sai}
\setcounterpageref{ex}{0}

\begin{ex} %[2D4H21]
Xét tính đúng sai của các phép cộng có nhớ:
\choiceTF
{\True $39 + 45 = 84$.}
{\True $58 + 26 = 84$.}
{$47 + 38 = 86$.}
{\True $47 + 38 = 85$.}
\loigiai{
\begin{itemize}
    \item $39 + 45 = 84$: Đúng.
    \item $58 + 26 = 84$: Đúng.
    \item $47 + 38 = 85 \neq 86$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\begin{ex} %[2D4V21]
Vườn có 46 cây xoài và 38 cây bưởi. Xét tính đúng sai:
\choiceTF
{\True Vườn có tất cả 84 cây ăn quả ($46 + 38 = 84$).}
{\True Số xoài nhiều hơn số bưởi là 8 cây ($46 - 38 = 8$).}
{Nếu trồng thêm 10 cây xoài thì vườn có 90 cây.}
{\True Khi đó vườn có $84 + 10 = 94$ cây.}
\loigiai{
\begin{itemize}
    \item Tổng: $46 + 38 = 84$: Đúng.
    \item Hiệu: $46 - 38 = 8$: Đúng.
    \item $84 + 10 = 94 \neq 90$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}
