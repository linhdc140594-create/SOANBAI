\textbf{Phần 1. Trắc nghiệm lựa chọn}
\setcounterpageref{ex}{0}

\begin{ex} %[2D10N52]
Số $456$ viết thành tổng các trăm, chục, đơn vị là
\choice
{$40 + 50 + 6$}
{$400 + 5 + 6$}
{\True $400 + 50 + 6$}
{$4 + 5 + 6$}
\loigiai{
$456 = 400 + 50 + 6$.\\
\textbf{Đáp án của câu hỏi là C}
}
\end{ex}

\textbf{Phần 2. Trắc nghiệm đúng sai}
\setcounterpageref{ex}{0}

\begin{ex} %[2D10H52]
Xét tính đúng sai khi phân tích số:
\choiceTF
{\True $548 = 500 + 40 + 8$.}
{\True $306 = 300 + 6$.}
{$720 = 700 + 2$.}
{\True $720 = 700 + 20$.}
\loigiai{
\begin{itemize}
    \item $548 = 500 + 40 + 8$: Đúng.
    \item $306 = 300 + 6$: Đúng.
    \item $720 = 700 + 20 \neq 700 + 2$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\begin{ex} %[2D10V52]
Cho số $M = 800 + 70 + 5$ và $N = 800 + 50 + 7$. Xét tính đúng sai:
\choiceTF
{\True $M = 875$.}
{\True $N = 857$.}
{$M < N$.}
{\True $M > N$ ($875 > 857$).}
\loigiai{
\begin{itemize}
    \item $M = 875$: Đúng.
    \item $N = 857$: Đúng.
    \item $M > N$ (không bé hơn). Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}
