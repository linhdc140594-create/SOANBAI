\setcounterpageref{ex}{0}

Điền dấu thích hợp: $300 \dots 500$.
\choice
{$>$}
{$=$}
{Không so sánh được}
{\True $<$}
\loigiai{
$300 < 500$.\\
\textbf{Đáp án của câu hỏi là D}

\end{ex}

\textbf{Phần 2. Trắc nghiệm đúng sai}
\setcounterpageref{ex}{0}

\begin{ex} %[2D10H50]
Xét tính đúng sai của các phép so sánh:
\choiceTF
{\True $600 > 400$.}
{\True $300 < 800$.}
{$500 = 50$.}
{\True $500 > 50$.}
\loigiai{
\begin{itemize}
    \item $600 > 400$: Đúng.
    \item $300 < 800$: Đúng.
    \item $500 > 50 \neq 50$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\begin{ex} %[2D10V50]
Cho 4 số: $200, 500, 700, 900$. Xét tính đúng sai:
\choiceTF
{\True Số lớn nhất là 900.}
{\True Số bé nhất là 200.}
{Tổng số bé nhất và số lớn nhất là 1 000.}
{\True Tổng số bé nhất và số lớn nhất là $200 + 900 = 1\,100$.}
\loigiai{
\begin{itemize}
    \item Lớn nhất là 900: Đúng.
    \item Bé nhất là 200: Đúng.
    \item $200 + 900 = 1100 \neq 1000$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\textbf{Phần 3. Trả lời ngắn}
\setcounterpageref{ex}{0}

\begin{ex} %[2D10H50]
Trong các số $400, 700, 200, 600$, tìm số lớn nhất.
\shortans{700}
\loigiai{
\textbf{Đáp số:} $700$\\
700.
}
\end{ex}

\begin{ex} %[2D10H50]
Trong các số $500, 800, 300, 900$, tìm số bé nhất.
\shortans{300}
\loigiai{
\textbf{Đáp số:} $300$\\
300.
}
\end{ex}

\begin{ex} %[2D10V50]
Tìm số tròn trăm $x$ biết $600 < x < 800$.
\shortans{700}
\loigiai{
\textbf{Đáp số:} $700$\\
700.
}
\end{ex}

\begin{ex} %[2D10V50]
Có bao nhiêu số tròn trăm nằm giữa 100 và 500?
\shortans{3}
\loigiai{
\textbf{Đáp số:} $3$\\
Gồm 200, 300, 400 (có 3 số).
}
\end{ex}

\begin{ex} %[2D10V50]
Tìm hiệu giữa số tròn trăm lớn nhất có 3 chữ số và số tròn trăm bé nhất có 3 chữ số.
\shortans{800}
\loigiai{
\textbf{Đáp số:} $800$\\
$900 - 100 = 800$.
}
\end{ex}

\begin{ex} %[2D10C50]
Tìm tổng của tất cả các số tròn trăm có ba chữ số lớn hơn 600 và nhỏ hơn 1 000. (Gồm 700, 800, 900)
\shortans{2400}
\loigiai{
\textbf{Đáp số:} $2400$\\
$700 + 800 + 900 = 2400$.
}
\end{ex}
