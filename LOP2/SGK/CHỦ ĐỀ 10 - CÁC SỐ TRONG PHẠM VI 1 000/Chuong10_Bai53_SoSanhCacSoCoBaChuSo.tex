\textbf{Phần 1. Trắc nghiệm lựa chọn}
\setcounterpageref{ex}{0}

\begin{ex} %[2D10N53]
Điền dấu thích hợp: $456 \dots 465$.
\choice
{$>$}
{\True $<$}
{$=$}
{Không so sánh được}
\loigiai{
Hàng trăm bằng 4. Hàng chục: $5 < 6$ nên $456 < 465$.\\
\textbf{Đáp án của câu hỏi là B}
}
\end{ex}

\textbf{Phần 2. Trắc nghiệm đúng sai}
\setcounterpageref{ex}{0}

\begin{ex} %[2D10H53]
Xét tính đúng sai của các so sánh số có 3 chữ số:
\choiceTF
{\True $435 < 453$.}
{\True $702 < 720$.}
{$890 < 809$.}
{\True $890 > 809$.}
\loigiai{
\begin{itemize}
    \item $435 < 453$: Đúng.
    \item $702 < 720$: Đúng.
    \item $890 > 809 \neq 809$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\begin{ex} %[2D10V53]
Cho dãy số: $350, 355, 360, 365, 370$. Xét tính đúng sai:
\choiceTF
{\True Đây là dãy số tăng dần đều 5 đơn vị.}
{\True Số lớn nhất trong dãy là 370.}
{Số bé nhất trong dãy là 360.}
{\True Số bé nhất trong dãy là 350.}
\loigiai{
\begin{itemize}
    \item Dãy tăng 5 đơn vị: Đúng.
    \item Lớn nhất là 370: Đúng.
    \item Bé nhất là 350 (không phải 360). Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\textbf{Phần 3. Trả lời ngắn}
\setcounterpageref{ex}{0}

\begin{ex} %[2D10H53]
Trong các số $672, 726, 267, 627$, tìm số lớn nhất.
\shortans{726}
\loigiai{
\textbf{Đáp số:} $726$\\
726.
}
\end{ex}

\begin{ex} %[2D10H53]

\begin{ex} %[2D10V53]
