\textbf{Phần 1. Trắc nghiệm lựa chọn}
\setcounterpageref{ex}{0}

\begin{ex} %[2D10N48]
10 đơn vị bằng mấy chục?
\choice
{$2$ chục}
{$10$ chục}
{$1$ trăm}
{\True $1$ chục}
\loigiai{
10 đơn vị = 1 chục.\\
\textbf{Đáp án của câu hỏi là D}
}
\end{ex}

\begin{ex} %[2D10N48]
10 chục bằng mấy trăm?
\choice
{$10$ trăm}
{\True $1$ trăm}
{$1$ nghìn}
{$2$ trăm}
\loigiai{
10 chục = 1 trăm (100).\\
\textbf{Đáp án của câu hỏi là B}
}
\end{ex}

\begin{ex} %[2D10H48]
10 trăm bằng mấy nghìn?
\choice
{$10$ nghìn}
{$100$ nghìn}
{\True $1$ nghìn ($1\,000$)}
{$2$ nghìn}
\loigiai{
10 trăm = 1 nghìn ($1000$).\\
\textbf{Đáp án của câu hỏi là C}
}
\end{ex}

\begin{ex} %[2D10H48]

\choice
{$100$}

{\True $1\,000$}
\loigiai{

\textbf{Đáp án của câu hỏi là D}
}
\end{ex}

\textbf{Phần 2. Trắc nghiệm đúng sai}
\setcounterpageref{ex}{0}

\begin{ex} %[2D10H48]
Xét mối quan hệ giữa các hàng đơn vị, chục, trăm, nghìn:
\choiceTF
{\True 10 đơn vị = 1 chục.}
{\True 10 chục = 1 trăm.}
{10 trăm = 10 nghìn.}
{\True 10 trăm = 1 nghìn ($1\,000$).}
\loigiai{
\begin{itemize}
    \item 10 đơn vị = 1 chục: Đúng.
    \item 10 chục = 1 trăm: Đúng.
    \item 10 trăm = 1 nghìn (không phải 10 nghìn). Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\begin{ex} %[2D10V48]
Cho số 1 000. Xét tính đúng sai:
\choiceTF
{\True 1 000 là số có bốn chữ số.}
{\True 1 000 là số liền sau của số 999.}
{1 000 là số lớn nhất có 3 chữ số.}
{\True 1 000 là số bé nhất có 4 chữ số.}
\loigiai{
\begin{itemize}
    \item 1000 có 4 chữ số: Đúng.
    \item Liền sau của 999: Đúng.
    \item 1000 là số có 4 chữ số (không phải 3 chữ số). Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\textbf{Phần 3. Trả lời ngắn}
\setcounterpageref{ex}{0}

\begin{ex} %[2D10H48]
10 chục bằng bao nhiêu trăm?
\shortans{1}
\loigiai{
\textbf{Đáp số:} $1$\\
1 trăm.
}
\end{ex}

\begin{ex} %[2D10H48]
10 trăm bằng bao nhiêu nghìn?
\shortans{1}
\loigiai{
\textbf{Đáp số:} $1$\\
1 nghìn.
}
\end{ex}

\begin{ex} %[2D10V48]
Viết số một nghìn bằng chữ số.
\shortans{1000}
\loigiai{
\textbf{Đáp số:} $1000$\\
1000.
}
\end{ex}

\begin{ex} %[2D10V48]
Tìm số liền trước của số $1\,000$.
\shortans{999}
\loigiai{
\textbf{Đáp số:} $999$\\
$1000 - 1 = 999$.
}
\end{ex}

\begin{ex} %[2D10V48]
Tìm số liền sau của số $999$.
\shortans{1000}
\loigiai{
\textbf{Đáp số:} $1000$\\
$999 + 1 = 1000$.
}
\end{ex}

\begin{ex} %[2D10C48]
Có bao nhiêu chữ số 0 trong số một nghìn ($1\,000$)?
\shortans{3}
\loigiai{
\textbf{Đáp số:} $3$\\
Có 3 chữ số 0.
}
\end{ex}
