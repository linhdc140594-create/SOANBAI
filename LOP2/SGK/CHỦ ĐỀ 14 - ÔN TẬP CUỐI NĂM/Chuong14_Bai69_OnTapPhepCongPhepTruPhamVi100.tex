\textbf{Phần 2. Trắc nghiệm đúng sai}
\setcounterpageref{ex}{0}

\begin{ex} %[2D14H69]
Xét tính đúng sai của các phép toán trong phạm vi 100:
\choiceTF
{\True $47 + 38 = 85$.}
{\True $91 - 46 = 45$.}
{$63 - 28 = 45$.}
{\True $63 - 28 = 35$.}
\loigiai{
\begin{itemize}
    \item $47 + 38 = 85$: Đúng.
    \item $91 - 46 = 45$: Đúng.
    \item $63 - 28 = 35 \neq 45$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\begin{ex} %[2D14V69]
Bạn Mai có 45 cái kẹo, bạn Lan có ít hơn bạn Mai 12 cái kẹo. Xét tính đúng sai:
\choiceTF
{\True Lan có 33 cái kẹo ($45 - 12 = 33$).}
{\True Cả hai bạn có tất cả 78 cái kẹo ($45 + 33 = 78$).}
{Cả hai bạn có tất cả 80 cái kẹo.}
{\True Nếu Lan được cho thêm 2 cái kẹo thì cả hai bạn có 80 cái kẹo.}
\loigiai{
\begin{itemize}
    \item Lan có 33 cái: Đúng.
    \item Cả 2 bạn có 78 cái: Đúng.
    \item $78 \neq 80$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\textbf{Phần 3. Trả lời ngắn}
\setcounterpageref{ex}{0}

\begin{ex} %[2D14H69]
Tính: $48 + 37$.
\shortans{85}
\loigiai{
\textbf{Đáp số:} $85$\\
$48 + 37 = 85$.
}
\end{ex}

\begin{ex} %[2D14H69]
Tính: $82 - 45$.
\shortans{37}
\loigiai{
\textbf{Đáp số:} $37$\\
$82 - 45 = 37$.
}
\end{ex}

\begin{ex} %[2D14V69]
Tính giá trị biểu thức: $100 - 36 - 24$.
\shortans{40}
\loigiai{
\textbf{Đáp số:} $40$\\
$64 - 24 = 40$.
}
\end{ex}

\begin{ex} %[2D14V69]
Tìm $x$ biết: $x + 38 = 91$.
\shortans{53}
\loigiai{
\textbf{Đáp số:} $53$\\
$x = 91 - 38 = 53$.
}
\end{ex}

\begin{ex} %[2D14V69]
Một cửa hàng có $80\text{ kg}$ gạo, buổi sáng bán được $35\text{ kg}$, buổi chiều bán được $28\text{ kg}$. Hỏi cửa hàng còn lại bao nhiêu ki-lô-gam gạo?
\shortans{17}
\loigiai{
\textbf{Đáp số:} $17$\\
$80 - (35 + 28) = 80 - 63 = 17\text{ kg}$.
}
\end{ex}

\begin{ex} %[2D14C69]
Tính nhanh: $27 + 38 + 13 + 12$.
\shortans{90}
\loigiai{
\textbf{Đáp số:} $90$\\
$(27 + 13) + (38 + 12) = 40 + 50 = 90$.
}
\end{ex}
