\textbf{Phần 2. Trắc nghiệm đúng sai}
\setcounterpageref{ex}{0}

\begin{ex} %[2D1H05]
Xét tính đúng sai của các phép tính:
\choiceTF
{\True $25 + 34 = 59$.}
{\True $78 - 45 = 33$.}
{$60 + 30 = 80$.}
{\True $60 + 30 = 90$.}
\loigiai{
\begin{itemize}
    \item $25 + 34 = 59$: Đúng.
    \item $78 - 45 = 33$: Đúng.
    \item $60 + 30 = 90 \neq 80$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\begin{ex} %[2D1V05]
Một cửa hàng buổi sáng bán được 35 gói kẹo, buổi chiều bán được 24 gói kẹo. Xét tính đúng sai:
\choiceTF
{\True Cả ngày bán được 59 gói kẹo ($35 + 24 = 59$).}
{\True Buổi sáng bán nhiều hơn buổi chiều 11 gói kẹo ($35 - 24 = 11$).}
{Nếu buổi chiều bán thêm 5 gói kẹo thì tổng số kẹo bán trong ngày là 65 gói.}
{\True Khi đó cả ngày bán được $59 + 5 = 64$ gói kẹo.}
\loigiai{
\begin{itemize}
    \item Tổng: $35 + 24 = 59$: Đúng.
    \item Hiệu: $35 - 24 = 11$: Đúng.
    \item $59 + 5 = 64 \neq 65$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\textbf{Phần 3. Trả lời ngắn}
\setcounterpageref{ex}{0}

\begin{ex} %[2D1H05]
Tính tổng: $42 + 35$.
\shortans{77}
\loigiai{
\textbf{Đáp số:} $77$\\
$42 + 35 = 77$.
}
\end{ex}

\begin{ex} %[2D1H05]
Tính hiệu: $89 - 46$.
\shortans{43}
\loigiai{
\textbf{Đáp số:} $43$\\
$89 - 46 = 43$.
}
\end{ex}

\begin{ex} %[2D1V05]
Tính giá trị biểu thức: $30 + 50 - 20$.
\shortans{60}
\loigiai{
\textbf{Đáp số:} $60$\\
$80 - 20 = 60$.
}
\end{ex}

\begin{ex} %[2D1V05]

\begin{ex} %[2D1V05]

\shortans{77}
\loigiai{
\textbf{Đáp số:} $77$\\
$45 + 32 = 77$ cây.
}
\end{ex}
