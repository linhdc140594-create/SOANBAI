\textbf{Phần 1. Trắc nghiệm lựa chọn}
\setcounterpageref{ex}{0}

\begin{ex} %[2H6N32]
Một tuần lễ gồm có mấy ngày?
\choice
{\True $7$ ngày}
{$6$ ngày}
{$8$ ngày}
{$5$ ngày}
\loigiai{
Một tuần có 7 ngày.\\
\textbf{Đáp án của câu hỏi là A}
}
\end{ex}

\textbf{Phần 2. Trắc nghiệm đúng sai}
\setcounterpageref{ex}{0}

\begin{ex} %[2H6H32]
Xét tính đúng sai về thời gian:
\choiceTF
{\True 22 giờ chính là 10 giờ đêm ($22 - 12 = 10$).}
{\True Tháng 7 có 31 ngày.}
{Tháng 6 có 31 ngày.}
{\True Tháng 6 chỉ có 30 ngày.}
\loigiai{
\begin{itemize}
    \item 22 giờ = 10 giờ đêm: Đúng.
    \item Tháng 7 có 31 ngày: Đúng.
    \item Tháng 6 có 30 ngày (không phải 31). Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\begin{ex} %[2H6V32]
Hôm nay là Chủ nhật ngày 16. Xét tính đúng sai:
\choiceTF
{\True Chủ nhật tuần sau là ngày 23 ($16 + 7 = 23$).}
{\True Chủ nhật tuần trước là ngày 9 ($16 - 7 = 9$).}
{Thứ Hai ngày mai là ngày 18.}
{\True Thứ Hai ngày mai là ngày 17 ($16 + 1 = 17$).}
\loigiai{
\begin{itemize}
    \item Tuần sau: $16 + 7 = 23$: Đúng.
    \item Tuần trước: $16 - 7 = 9$: Đúng.
    \item $16 + 1 = 17 \neq 18$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\textbf{Phần 3. Trả lời ngắn}
\setcounterpageref{ex}{0}

\begin{ex} %[2H6H32]
Một tuần lễ có bao nhiêu ngày?
\shortans{7}
\loigiai{
\textbf{Đáp số:} $7$\\
7 ngày.
}
\end{ex}

\begin{ex} %[2H6H32]
Tháng 10 có bao nhiêu ngày?
\shortans{31}
\loigiai{
\textbf{Đáp số:} $31$\\
31 ngày.
}
\end{ex}

\begin{ex} %[2H6V32]
Nếu thứ Ba là ngày 11 thì thứ Ba tuần sau là ngày bao nhiêu?
\shortans{18}
\loigiai{
\textbf{Đáp số:} $18$\\
$11 + 7 = 18$.
}
\end{ex}

\begin{ex} %[2H6V32]
Lúc 16 giờ chiều tương ứng với mấy giờ chiều?
\shortans{4}
\loigiai{
\textbf{Đáp số:} $4$\\
$16 - 12 = 4$ giờ chiều.
}
\end{ex}

\begin{ex} %[2H6V32]

$11 - 7 = 4$ giờ.
}
\end{ex}
