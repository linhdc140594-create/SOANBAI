{\footnotesize
\begin{itemize}\setlength{\itemsep}{1pt}\setlength{\leftskip}{-2.5mm}
  \item \textbf{Trường hợp số chữ số khác nhau:} Số nào có nhiều chữ số hơn thì \textbf{lớn hơn} (ví dụ: $1\,000 > 999$).
  \item \textbf{Trường hợp đều có 3 chữ số:} So sánh từng cặp chữ số ở cùng một hàng từ \textbf{trái sang phải}:
  \begin{itemize}\setlength{\itemsep}{0.5pt}\setlength{\leftskip}{-2mm}
    \item So sánh chữ số \textbf{hàng trăm} trước.
    \item Nếu hàng trăm bằng nhau, so sánh chữ số \textbf{hàng chục}.
    \item Nếu hàng chục cũng bằng nhau, so sánh chữ số \textbf{hàng đơn vị}.
    \item Nếu cả ba hàng đều bằng nhau thì hai số \textbf{bằng nhau}.
  \end{itemize}
\end{itemize}
}
\end{tcolorbox}

{\footnotesize
\begin{itemize}\setlength{\itemsep}{1pt}\setlength{\leftskip}{-2.5mm}
  \item \textbf{Ví dụ 1:} So sánh $537$ và $489$:
  \begin{itemize}\setlength{\itemsep}{0.5pt}\setlength{\leftskip}{-2mm}
    \item Hàng trăm: $5 > 4 \implies \mathbf{537 > 489}$.
  \end{itemize}
  \item \textbf{Ví dụ 2:} So sánh $642$ và $671$:
  \begin{itemize}\setlength{\itemsep}{0.5pt}\setlength{\leftskip}{-2mm}
    \item Hàng trăm: $6 = 6$.
    \item Hàng chục: $4 < 7 \implies \mathbf{642 < 671}$.
  \end{itemize}
  \item \textbf{Ví dụ 3:} So sánh $854$ và $859$:
  \begin{itemize}\setlength{\itemsep}{0.5pt}\setlength{\leftskip}{-2mm}
    \item Hàng trăm: $8 = 8$, hàng chục: $5 = 5$.
    \item Hàng đơn vị: $4 < 9 \implies \mathbf{854 < 859}$.
  \end{itemize}
\end{itemize}
}

{\footnotesize
Cho các số: $452, 389, 471, 365$.\\
\textbf{1. So sánh từng số:}\\
- Các số hàng trăm là $3$: $365 < 389$.\\
- Các số hàng trăm là $4$: $452 < 471$.\\
\textbf{2. Kết quả sắp xếp:}\\
- \textbf{Từ bé đến lớn:} $365,\ 389,\ 452,\ 471$.\\
- \textbf{Từ lớn đến bé:} $471,\ 452,\ 389,\ 365$.\\
- Số bé nhất là: $\mathbf{365}$; Số lớn nhất là: $\mathbf{471}$.
}
