  \item \textbf{Số đặc biệt:}
  \begin{itemize}\setlength{\itemsep}{0.5pt}\setlength{\leftskip}{-2mm}
    \item Số bé nhất có 3 chữ số: $\mathbf{100}$.
    \item Số lớn nhất có 3 chữ số: $\mathbf{999}$.
    \item Số tròn trăm bé nhất: $100$; lớn nhất: $900$.
  \end{itemize}

  \item \textbf{Số liền trước - liền sau:}
  \begin{itemize}\setlength{\itemsep}{0.5pt}\setlength{\leftskip}{-2mm}
    \item Liền trước $n$ là $n - 1$ (liền trước $1\,000$ là $999$).
    \item Liền sau $n$ là $n + 1$ (liền sau $999$ là $1\,000$).
  \end{itemize}
  \item \textbf{Quy luật dãy số thường gặp:}
  \begin{itemize}\setlength{\itemsep}{0.5pt}\setlength{\leftskip}{-2mm}
    \item Đếm thêm $1$, đếm thêm $10$ (tròn chục), đếm thêm $100$ (tròn trăm).
  \end{itemize}
\end{itemize}
