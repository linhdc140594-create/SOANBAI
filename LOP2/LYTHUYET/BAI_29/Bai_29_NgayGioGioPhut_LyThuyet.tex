  \item \textbf{Mẹo đổi giờ chiều sang 24h:}\\
  Lấy số giờ chiều \textbf{cộng thêm 12}:\\
  $4\text{ giờ chiều} \implies 4 + 12 = 16\text{ giờ}$.
  \item \textbf{Mẹo đổi từ 24h sang giờ chiều/tối:}\\
  Lấy số giờ trừ đi 12:\\
  $18\text{ giờ} \implies 18 - 12 = 6\text{ giờ tối}$.
  \item Không nhầm kim ngắn (giờ) và kim dài (phút).
\end{itemize}
