\begin{itemize}\setlength{\itemsep}{1pt}\setlength{\leftskip}{-2.5mm}
  \item \textbf{Bảng cộng, trừ qua 10:} Thuộc lòng các bảng cộng, trừ cơ bản ($9+5=14, 8+7=15, 13-6=7, 15-8=7\dots$).
  \item \textbf{Phép cộng có nhớ:}\\
  Cộng hàng đơn vị $\ge 10 \implies$ viết hàng đơn vị, \textbf{nhớ $1$} sang hàng chục.
  \item \textbf{Phép trừ có nhớ:}\\
  Hàng đơn vị không trừ được $\implies$ mượn $1$ chục ($10$), trừ rồi \textbf{trả $1$} vào số trừ hàng chục.
\end{itemize}

{\footnotesize
\begin{itemize}\setlength{\itemsep}{1pt}\setlength{\leftskip}{-2.5mm}
  \item \textbf{Trong phép cộng:}
  $$\text{Số hạng} + \text{Số hạng} = \text{Tổng}$$
  $$\implies \mathbf{\text{Số hạng chưa biết} = \text{Tổng} - \text{Số hạng đã biết}}$$
  \item \textbf{Trong phép trừ:}
  $$\text{Số bị trừ} - \text{Số trừ} = \text{Hiệu}$$
  $$\implies \mathbf{\text{Số bị trừ} = \text{Hiệu} + \text{Số trừ}}$$
  $$\implies \mathbf{\text{Số trừ} = \text{Số bị trừ} - \text{Hiệu}}$$
\end{itemize}
}
\end{tcolorbox}
