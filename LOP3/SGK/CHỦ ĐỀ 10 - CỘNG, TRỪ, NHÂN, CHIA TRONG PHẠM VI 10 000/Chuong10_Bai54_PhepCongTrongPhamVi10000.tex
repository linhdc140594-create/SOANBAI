\textbf{Phần 2. Trắc nghiệm đúng sai}
\setcounterpageref{ex}{0}

\begin{ex} %[3D10H54]
Xét tính đúng sai của các phép cộng:
\choiceTF
{\True $3200 + 1800 = 5000$.}
{\True $4560 + 2340 = 6900$.}
{$5432 + 1234 = 6656$.}
{\True $5432 + 1234 = 6666$.}
\loigiai{
\begin{itemize}
    \item $3200 + 1800 = 5000$: Đúng.
    \item $4560 + 2340 = 6900$: Đúng.
    \item $5432 + 1234 = 6666 \neq 6656$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\begin{ex} %[3D10V54]
Một nông trại thu hoạch được $3\,450\text{ kg}$ bắp cải và $2\,850\text{ kg}$ su hào. Xét tính đúng sai:
\choiceTF
{\True Tổng khối lượng rau củ thu hoạch là $6\,300\text{ kg}$.}
{\True Khối lượng bắp cải nhiều hơn su hào là $600\text{ kg}$.}
{Nếu thu hoạch thêm $700\text{ kg}$ bắp cải thì tổng số rau là $7\,100\text{ kg}$.}
{\True Nếu thu hoạch thêm $700\text{ kg}$ bắp cải thì tổng số rau là $7\,000\text{ kg}$ ($6300 + 700 = 7000$).}
\loigiai{
\begin{itemize}
    \item Tổng: $3450 + 2850 = 6300\text{ kg}$. Đúng.
    \item Hiệu: $3450 - 2850 = 600\text{ kg}$. Đúng.
    \item $6300 + 700 = 7000\text{ kg} \neq 7100\text{ kg}$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\textbf{Phần 3. Trả lời ngắn}

\begin{ex} %[3D10H54]
Tính tổng: $4\,325 + 2\,431$.
\shortans{6756}
\loigiai{
\textbf{Đáp số:} $6756$\\
$4325 + 2431 = 6756$.
}
\end{ex}

\begin{ex} %[3D10H54]

\begin{ex} %[3D10V54]
