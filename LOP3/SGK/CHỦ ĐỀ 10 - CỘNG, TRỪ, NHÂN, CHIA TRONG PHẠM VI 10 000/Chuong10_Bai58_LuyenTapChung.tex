\textbf{Phần 2. Trắc nghiệm đúng sai}
\setcounterpageref{ex}{0}

\begin{ex} %[3D10H58]
Xét tính đúng sai của các biểu thức tính toán:
\choiceTF
{\True $2000 \times 3 + 1000 = 7000$.}
{\True $8000 : 2 - 1500 = 2500$.}
{$(3000 + 1000) \times 2 = 6000$.}
{\True $(3000 + 1000) \times 2 = 8000$.}
\loigiai{
\begin{itemize}
    \item $6000 + 1000 = 7000$: Đúng.
    \item $4000 - 1500 = 2500$: Đúng.
    \item $4000 \times 2 = 8000 \neq 6000$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\begin{ex} %[3D10V58]
Một kho gạo có $8\,400\text{ kg}$ gạo. Lần 1 chuyển đi $2\,500\text{ kg}$, lần 2 chuyển đi $3\,500\text{ kg}$. Xét tính đúng sai:
\choiceTF
{\True Cả 2 lần chuyển được tất cả $6\,000\text{ kg}$ gạo.}
{\True Số gạo còn lại trong kho là $2\,400\text{ kg}$.}
{Lần 2 chuyển nhiều hơn lần 1 là $800\text{ kg}$.}
{\True Lần 2 chuyển nhiều hơn lần 1 là $1\,000\text{ kg}$ ($3500 - 2500 = 1000$).}
\loigiai{
\begin{itemize}
    \item Đã chuyển: $2500 + 3500 = 6000\text{ kg}$. Đúng.
    \item Còn lại: $8400 - 6000 = 2400\text{ kg}$. Đúng.
    \item Hiệu: $3500 - 2500 = 1000\text{ kg} \neq 800$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\textbf{Phần 3. Trả lời ngắn}
\setcounterpageref{ex}{0}

\begin{ex} %[3D10H58]
Tính: $3\,450 + 4\,550$.
\shortans{8000}
\loigiai{
\textbf{Đáp số:} $8000$\\
$3450 + 4550 = 8000$.
}
\end{ex}

\begin{ex} %[3D10H58]

\shortans{5550}
\loigiai{
\textbf{Đáp số:} $5550$\\
$9000 - 3450 = 5550$.
}
\end{ex}

\begin{ex} %[3D10V58]

\loigiai{
\textbf{Đáp số:} $5800$\\

\end{ex}

\begin{ex} %[3D10V58]
