\textbf{Phần 2. Trắc nghiệm đúng sai}
\setcounterpageref{ex}{0}

\begin{ex} %[3D10H55]
Xét tính đúng sai của các phép trừ:
\choiceTF
{\True $6000 - 2500 = 3500$.}
{\True $8740 - 3420 = 5320$.}
{$10000 - 3200 = 6700$.}
{\True $10000 - 3200 = 6800$.}
\loigiai{
\begin{itemize}
    \item $6000 - 2500 = 3500$: Đúng.
    \item $8740 - 3420 = 5320$: Đúng.
    \item $10000 - 3200 = 6800 \neq 6700$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\begin{ex} %[3D10V55]
Một cuộn dây điện dài $7\,500\text{ m}$. Lần 1 cắt $2\,200\text{ m}$, lần 2 cắt $2\,800\text{ m}$. Xét tính đúng sai:
\choiceTF
{\True Cả 2 lần đã cắt được $5\,000\text{ m}$ dây điện.}
{\True Cuộn dây còn lại $2\,500\text{ m}$.}
{Lần 2 cắt ít hơn lần 1 là $600\text{ m}$.}
{\True Lần 2 cắt nhiều hơn lần 1 là $600\text{ m}$ ($2800 - 2200 = 600$).}
\loigiai{
\begin{itemize}
    \item Đã cắt: $2200 + 2800 = 5000\text{ m}$. Đúng.
    \item Còn lại: $7500 - 5000 = 2500\text{ m}$. Đúng.
    \item Lần 2 ($2800$) nhiều hơn lần 1 ($2200$). Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\textbf{Phần 3. Trả lời ngắn}
\setcounterpageref{ex}{0}

\begin{ex} %[3D10H55]
Tính hiệu: $7\,856 - 3\,421$.
\shortans{4435}
\loigiai{
\textbf{Đáp số:} $4435$\\
$7856 - 3421 = 4435$.
}
\end{ex}

\begin{ex} %[3D10H55]
Tính hiệu: $9\,430 - 4\,685$.
\shortans{4745}
\loigiai{
\textbf{Đáp số:} $4745$\\
$9430 - 4685 = 4745$.
}
\end{ex}

\begin{ex} %[3D10V55]
Tính: $10\,000 - 6\,350$.
\shortans{3650}
\loigiai{
\textbf{Đáp số:} $3650$\\
$10000 - 6350 = 3650$.
}
\end{ex}

\begin{ex} %[3D10V55]
Tìm $x$ biết: $7\,500 - x = 2\,800$.
\shortans{4700}
\loigiai{
\textbf{Đáp số:} $4700$\\
$x = 7500 - 2800 = 4700$.
}
\end{ex}

\begin{ex} %[3D10V55]
Một bể nước chứa $5\,000\text{ l}$ nước. Sau khi dùng hết $3\,450\text{ l}$ thì trong bể còn lại bao nhiêu lít nước?
\shortans{1550}
\loigiai{
\textbf{Đáp số:} $1550$\\
$5000 - 3450 = 1550\text{ l}$.
}
\end{ex}

\begin{ex} %[3D10C55]
Tính nhanh: $8\,650 - 2\,300 - 1\,350$.
\shortans{5000}
\loigiai{
\textbf{Đáp số:} $5000$\\
$(8650 - 1350) - 2300 = 7300 - 2300 = 5000$.
}
\end{ex}
