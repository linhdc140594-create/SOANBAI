\begin{ex} %[3D10C57]
Một kho hàng có $9\,600\text{ kg}$ ngô. Lần đầu xuất đi $\dfrac{1}{3}$ số ngô, lần sau xuất đi $\dfrac{1}{4}$ số ngô CÒN LẠI. Số ngô còn lại cuối cùng là
\choice
{\True $4800\text{ kg}$}
{$4600\text{ kg}$}
{$5000\text{ kg}$}
{$5200\text{ kg}$}
\loigiai{
Lần 1 xuất: $9600 : 3 = 3200\text{ kg}$.\\
Còn lại sau lần 1: $9600 - 3200 = 6400\text{ kg}$.\\
Lần 2 xuất: $6400 : 4 = 1600\text{ kg}$.\\
Còn lại: $6400 - 1600 = 4800\text{ kg}$.\\
\textbf{Đáp án của câu hỏi là A}
}
\end{ex}

\textbf{Phần 2. Trắc nghiệm đúng sai}
\setcounterpageref{ex}{0}

\begin{ex} %[3D10H57]
Xét tính đúng sai của các phép chia:
\choiceTF
{\True $6482 : 2 = 3241$.}
{\True $9369 : 3 = 3123$.}
{$5625 : 5 = 125$.}
{\True $5625 : 5 = 1125$.}
\loigiai{
\begin{itemize}
    \item $6482 : 2 = 3241$: Đúng.
    \item $9369 : 3 = 3123$: Đúng.
    \item $5625 : 5 = 1125 \neq 125$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\begin{ex} %[3D10V57]
Một xưởng sản xuất được $4\,896$ chiếc khẩu trang đóng đều vào 4 hộp lớn. Xét tính đúng sai:
\choiceTF
{\True Mỗi hộp lớn có $1\,224$ chiếc khẩu trang.}
{\True 2 hộp lớn có $2\,448$ chiếc khẩu trang.}
{3 hộp lớn có $3\,662$ chiếc khẩu trang.}
{\True 3 hộp lớn có $3\,672$ chiếc khẩu trang ($1224 \times 3 = 3672$).}
\loigiai{
\begin{itemize}
    \item Mỗi hộp: $4896 : 4 = 1224$ chiếc. Đúng.
    \item 2 hộp: $1224 \times 2 = 2448$ chiếc. Đúng.
    \item 3 hộp: $1224 \times 3 = 3672 \neq 3662$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}
