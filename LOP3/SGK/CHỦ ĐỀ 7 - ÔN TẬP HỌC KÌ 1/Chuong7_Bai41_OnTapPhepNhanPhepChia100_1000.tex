\begin{ex} %[3D7C41]
Một kho thóc có $720\text{ kg}$ thóc. Lần đầu xuất đi $\dfrac{1}{6}$ số thóc, lần sau xuất đi $\dfrac{1}{5}$ số thóc CÒN LẠI. Số thóc còn lại trong kho là
\choice
{$460\text{ kg}$}
{$500\text{ kg}$}
{\True $480\text{ kg}$}
{$520\text{ kg}$}
\loigiai{
Lần đầu xuất: $720 : 6 = 120\text{ kg}$.\\
Còn lại sau lần 1: $720 - 120 = 600\text{ kg}$.\\
Lần sau xuất: $600 : 5 = 120\text{ kg}$.\\
Còn lại cuối cùng: $600 - 120 = 480\text{ kg}$.\\
\textbf{Đáp án của câu hỏi là C}
}
\end{ex}

\textbf{Phần 2. Trắc nghiệm đúng sai}
\setcounterpageref{ex}{0}

\begin{ex} %[3D7H41]
Xét tính đúng sai của các phép toán:
\choiceTF
{\True $7 \times 9 = 63$.}
{\True $81 : 9 = 9$.}
{$115 \times 4 = 450$.}
{\True $115 \times 4 = 460$.}
\loigiai{
\begin{itemize}
    \item $7 \times 9 = 63$: Đúng.
    \item $81 : 9 = 9$: Đúng.
    \item $115 \times 4 = 460 \neq 450$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\begin{ex} %[3D7V41]
Có 420 quyển sách xếp đều vào 4 ngăn. Xét tính đúng sai:
\choiceTF
{\True Mỗi ngăn có 105 quyển sách.}
{\True 2 ngăn có tất cả 210 quyển sách.}
{3 ngăn có tất cả 310 quyển sách.}
{\True 3 ngăn có tất cả 315 quyển sách.}
\loigiai{
\begin{itemize}
    \item Mỗi ngăn: $420 : 4 = 105$ quyển. Đúng.
    \item 2 ngăn: $105 \times 2 = 210$ quyển. Đúng.
    \item 3 ngăn: $105 \times 3 = 315$ quyển $\neq 310$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\textbf{Phần 3. Trả lời ngắn}
\setcounterpageref{ex}{0}

\begin{ex} %[3D7H41]
Tính tích: $114 \times 3$.
\shortans{342}
\loigiai{
\textbf{Đáp số:} $342$\\
$114 \times 3 = 342$.
}
\end{ex}

\begin{ex} %[3D7H41]
Tính thương: $555 : 5$.
\shortans{111}
\loigiai{
\textbf{Đáp số:} $111$\\
$555 : 5 = 111$.
}
\end{ex}

\begin{ex} %[3D7V41]
