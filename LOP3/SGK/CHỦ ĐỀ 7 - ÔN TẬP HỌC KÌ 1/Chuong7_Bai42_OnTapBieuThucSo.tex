\begin{ex} %[3D7N42]

\choice
{$60$}
{$80$}
{\True $70$}
{$50$}
\loigiai{
$40 + 60 - 30 = 100 - 30 = 70$.\\
\textbf{Đáp án của câu hỏi là C}
}

\textbf{Phần 2. Trắc nghiệm đúng sai}
\setcounterpageref{ex}{0}

\begin{ex} %[3D7H42]
Xét tính đúng sai của thứ tự thực hiện biểu thức:
\choiceTF
{\True $30 + 10 \times 3 = 60$.}
{\True $(30 + 10) \times 3 = 120$.}
{$80 - 40 : 4 = 10$.}
{\True $80 - 40 : 4 = 70$ (vì $80 - 10 = 70$).}
\loigiai{
\begin{itemize}
    \item $30 + 30 = 60$: Đúng.
    \item $40 \times 3 = 120$: Đúng.
    \item $80 - 10 = 70 \neq 10$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\begin{ex} %[3D7V42]
Cho 2 biểu thức: $M = (180 - 60) : 3$ và $N = 180 - 60 : 3$. Xét tính đúng sai:
\choiceTF
{\True Giá trị biểu thức $M$ là $40$.}
{\True Giá trị biểu thức $N$ là $160$.}
{Giá trị biểu thức $M$ lớn hơn giá trị biểu thức $N$.}
{\True Giá trị biểu thức $M$ bé hơn giá trị biểu thức $N$ ($40 < 160$).}
\loigiai{
\begin{itemize}
    \item $M = 120 : 3 = 40$. Đúng.
    \item $N = 180 - 20 = 160$. Đúng.
    \item $40 < 160$ nên $M < N$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\textbf{Phần 3. Trả lời ngắn}
\setcounterpageref{ex}{0}

\begin{ex} %[3D7H42]
Tính giá trị biểu thức: $50 + 25 \times 2$.
\shortans{100}
\loigiai{
\textbf{Đáp số:} $100$\\
$50 + 50 = 100$.
}
\end{ex}

\begin{ex} %[3D7H42]
Tính giá trị biểu thức: $(90 - 30) : 6$.
\shortans{10}
\loigiai{
\textbf{Đáp số:} $10$\\
$60 : 6 = 10$.
}
\end{ex}

\begin{ex} %[3D7V42]
Tính giá trị biểu thức: $200 - 80 : 4$.
\shortans{180}
\loigiai{
\textbf{Đáp số:} $180$\\
$200 - 20 = 180$.
}
\end{ex}

\begin{ex} %[3D7V42]
Tính giá trị biểu thức: $(15 + 25) \times 4$.
\shortans{160}
\loigiai{
\textbf{Đáp số:} $160$\\
$40 \times 4 = 160$.
}
\end{ex}

\begin{ex} %[3D7V42]
Tìm $x$ biết: $(x - 5) \times 3 = 45$.
\shortans{20}
\loigiai{
\textbf{Đáp số:} $20$\\
$x - 5 = 45 : 3 = 15 \Rightarrow x = 15 + 5 = 20$.
}
\end{ex}

\begin{ex} %[3D7C42]
Tính nhanh: $2 \times 45 \times 5$.
\shortans{450}
\loigiai{
\textbf{Đáp số:} $450$\\
$(2 \times 5) \times 45 = 10 \times 45 = 450$.
}
\end{ex}
