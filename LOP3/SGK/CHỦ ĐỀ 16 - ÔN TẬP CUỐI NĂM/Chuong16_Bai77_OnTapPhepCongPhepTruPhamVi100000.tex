{$80000$}

{$95000$}

\loigiai{
$50000 + 40000 = 90000$.\\
\textbf{Đáp án của câu hỏi là D}

\textbf{Phần 2. Trắc nghiệm đúng sai}
\setcounterpageref{ex}{0}

\begin{ex} %[3D16H77]
Xét tính đúng sai của các phép toán:
\choiceTF
{\True $60000 + 40000 = 100000$.}
{\True $100000 - 55000 = 45000$.}
{$45000 + 25000 - 30000 = 35000$.}
{\True $45000 + 25000 - 30000 = 40000$.}
\loigiai{
\begin{itemize}
    \item $60000 + 40000 = 100000$: Đúng.
    \item $100000 - 55000 = 45000$: Đúng.
    \item $70000 - 30000 = 40000 \neq 35000$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\begin{ex} %[3D16V77]
Một công ty dự định may $90\,000$ bộ quần áo. Tháng 1 may được $32\,000$ bộ, tháng 2 may được $38\,000$ bộ. Xét tính đúng sai:
\choiceTF
{\True Cả 2 tháng công ty may được $70\,000$ bộ.}
{\True Công ty còn phải may $20\,000$ bộ nữa.}
{Tháng 2 may nhiều hơn tháng 1 là $8\,000$ bộ.}
{\True Tháng 2 may nhiều hơn tháng 1 là $6\,000$ bộ ($38000 - 32000 = 6000$).}
\loigiai{
\begin{itemize}
    \item Tổng 2 tháng: $32000 + 38000 = 70000$ bộ. Đúng.
    \item Còn lại: $90000 - 70000 = 20000$ bộ. Đúng.
    \item $38000 - 32000 = 6000 \neq 8000$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\textbf{Phần 3. Trả lời ngắn}
\setcounterpageref{ex}{0}

\begin{ex} %[3D16H77]
Tính tổng: $54\,320 + 35\,680$.
\shortans{90000}
\loigiai{
\textbf{Đáp số:} $90000$\\
$54320 + 35680 = 90000$.
}
\end{ex}

\begin{ex} %[3D16H77]
Tính hiệu: $95\,000 - 48\,500$.
\shortans{46500}
\loigiai{
\textbf{Đáp số:} $46500$\\
$95000 - 48500 = 46500$.
}
\end{ex}

\begin{ex} %[3D16V77]
Tính giá trị biểu thức: $100\,000 - 35\,000 - 25\,000$.
\shortans{40000}
\loigiai{
\textbf{Đáp số:} $40000$\\
$100000 - 60000 = 40000$.
}
\end{ex}

\begin{ex} %[3D16V77]
Tìm $x$ biết: $x + 35\,000 = 90\,000$.
\shortans{55000}
\loigiai{
\textbf{Đáp số:} $55000$\\
$x = 90000 - 35000 = 55000$.
}
\end{ex}

\begin{ex} %[3D16V77]
Một kho thóc có $80\,000\text{ kg}$. Lần 1 xuất $25\,000\text{ kg}$, lần 2 xuất $35\,000\text{ kg}$. Trong kho còn lại bao nhiêu ki-lô-gam thóc?
\shortans{20000}
\loigiai{
\textbf{Đáp số:} $20000$\\
$80000 - (25000 + 35000) = 20000\text{ kg}$.
}
\end{ex}

\begin{ex} %[3D16C77]
Tính nhanh: $45\,000 + 26\,000 + 15\,000 - 16\,000$.
\shortans{70000}
\loigiai{
\textbf{Đáp số:} $70000$\\
$(45000 + 15000) + (26000 - 16000) = 60000 + 10000 = 70000$.
}
\end{ex}
