\textbf{Phần 1. Trắc nghiệm lựa chọn}
\setcounterpageref{ex}{0}

\begin{ex} %[3H16N79]
Chu vi hình vuông có cạnh $8\text{ cm}$ là
\choice
{$64\text{ cm}$}
{$24\text{ cm}$}
{\True $32\text{ cm}$}
{$16\text{ cm}$}
\loigiai{
$8 \times 4 = 32\text{ cm}$.\\
\textbf{Đáp án của câu hỏi là C}
}
\end{ex}

\textbf{Phần 2. Trắc nghiệm đúng sai}
\setcounterpageref{ex}{0}

\begin{ex} %[3H16H79]
Xét tính đúng sai về đơn vị đo:
\choiceTF
{\True $1\text{ km} = 1000\text{ m}$.}
{\True $1\text{ l} = 1000\text{ ml}$.}
{$1\text{ kg} = 100\text{ g}$.}
{\True $1\text{ kg} = 1000\text{ g}$.}
\loigiai{
\begin{itemize}
    \item $1\text{ km} = 1000\text{ m}$: Đúng.
    \item $1\text{ l} = 1000\text{ ml}$: Đúng.
    \item $1\text{ kg} = 1000\text{ g} \neq 100\text{ g}$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\begin{ex} %[3H16V79]
Một mảnh vườn hình chữ nhật có chiều dài $15\text{ m}$, chiều rộng $10\text{ m}$. Xét tính đúng sai:
\choiceTF
{\True Chu vi mảnh vườn là $50\text{ m}$.}
{\True Diện tích mảnh vườn là $150\text{ m}^2$.}
{Nếu tăng chiều rộng thêm $5\text{ m}$ thì mảnh vườn trở thành hình vuông có chu vi $60\text{ m}$.}
{\True Khi đó mảnh vườn thành hình vuông cạnh $15\text{ m}$, chu vi là $15 \times 4 = 60\text{ m}$.}
\loigiai{
\begin{itemize}
    \item Chu vi: $(15 + 10) \times 2 = 50\text{ m}$. Đúng.
    \item Diện tích: $15 \times 10 = 150\text{ m}^2$. Đúng.
    \item Trở thành hình vuông cạnh $15\text{ m}$, chu vi $60\text{ m}$. Cả hai ý cuối đều Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Đúng}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\textbf{Phần 3. Trả lời ngắn}
\setcounterpageref{ex}{0}

\begin{ex} %[3H16H79]
Tính chu vi hình vuông có cạnh dài $15\text{ cm}$.
\shortans{60}
\loigiai{
\textbf{Đáp số:} $60$\\
$15 \times 4 = 60\text{ cm}$.
}
\end{ex}

\begin{ex} %[3H16H79]
Tính diện tích hình chữ nhật có chiều dài $14\text{ cm}$, chiều rộng $5\text{ cm}$.
\shortans{70}
\loigiai{
\textbf{Đáp số:} $70$\\
$14 \times 5 = 70\text{ cm}^2$.
}
\end{ex}

\begin{ex} %[3H16V79]
Một hình vuông có diện tích $64\text{ cm}^2$. Tính chu vi của hình vuông đó theo xăng-ti-mét.
\shortans{32}
\loigiai{
\textbf{Đáp số:} $32$\\
Cạnh là $8\text{ cm}$. Chu vi: $8 \times 4 = 32\text{ cm}$.
}
\end{ex}

\begin{ex} %[3H16V79]
Tính chu vi hình chữ nhật có diện tích $45\text{ cm}^2$ và chiều rộng $5\text{ cm}$.
\shortans{28}
\loigiai{
\textbf{Đáp số:} $28$\\
Dài: $45 : 5 = 9\text{ cm}$. Chu vi: $(9 + 5) \times 2 = 28\text{ cm}$.
}
\end{ex}

\begin{ex} %[3H16V79]
Đổi: $4\text{ kg } 500\text{ g} = \dots\text{ g}$.
\shortans{4500}
\loigiai{
\textbf{Đáp số:} $4500$\\
$4000 + 500 = 4500\text{ g}$.
}
\end{ex}

\begin{ex} %[3H16C79]
Một hình chữ nhật có chu vi $32\text{ cm}$ và chiều dài gấp 3 lần chiều rộng. Tính diện tích hình chữ nhật theo xăng-ti-mét vuông.
\shortans{48}
\loigiai{
\textbf{Đáp số:} $48$\\
Nửa chu vi: $16\text{ cm}$. Rộng: $16 : 4 = 4\text{ cm}$, Dài: $12\text{ cm}$.\\
Diện tích: $12 \times 4 = 48\text{ cm}^2$.
}
\end{ex}
