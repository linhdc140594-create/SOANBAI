\textbf{Phần 1. Trắc nghiệm lựa chọn}
\setcounterpageref{ex}{0}

\begin{ex} %[3D6N40]
Kết quả của phép nhân $121 \times 4$ là
\choice
{$464$}
{$494$}
{\True $484$}
{$474$}
\loigiai{
$121 \times 4 = 484$.\\
\textbf{Đáp án của câu hỏi là C}
}
\end{ex}

\begin{ex} %[3D6V40]
Một hình chữ nhật có chu vi $60\text{ cm}$, chiều rộng là $10\text{ cm}$. Chiều dài gấp mấy lần chiều rộng?
\choice
{$1$ lần}
{$3$ lần}
{$4$ lần}
{\True $2$ lần}
\loigiai{
Nửa chu vi: $60 : 2 = 30\text{ cm}$.\\
Chiều dài: $30 - 10 = 20\text{ cm}$.\\
Số lần: $20 : 10 = 2$ lần.\\
\textbf{Đáp án của câu hỏi là D}
}
\end{ex}

\begin{ex} %[3D6C40]
Một thùng đựng $84\text{ l}$ dầu. Người ta đã bán đi $\dfrac{1}{4}$ số dầu, sau đó bán tiếp $15\text{ l}$ nữa. Số dầu còn lại trong thùng là
\choice
{$46\text{ l}$}
{\True $48\text{ l}$}
{$50\text{ l}$}
{$52\text{ l}$}
\loigiai{
Lần 1 bán: $84 : 4 = 21\text{ l}$.\\
Tổng đã bán: $21 + 15 = 36\text{ l}$.\\
Còn lại: $84 - 36 = 48\text{ l}$.\\
\textbf{Đáp án của câu hỏi là B}
}
\end{ex}

\textbf{Phần 2. Trắc nghiệm đúng sai}
\setcounterpageref{ex}{0}

\begin{ex} %[3D6H40]
Xét tính đúng sai của các phép toán:
\choiceTF
{\True $130 \times 3 = 390$.}
{\True $848 : 8 = 106$.}
{$50 + 50 \times 2 = 200$.}
{\True $50 + 50 \times 2 = 150$.}
\loigiai{
\begin{itemize}
    \item $130 \times 3 = 390$: Đúng.
    \item $848 : 8 = 106$: Đúng.
    \item $50 + 100 = 150 \neq 200$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\begin{ex} %[3D6V40]
Một cửa hàng có 3 bao gạo, mỗi bao nặng $120\text{ kg}$. Đã bán đi $\dfrac{1}{3}$ tổng số gạo. Xét tính đúng sai:
\choiceTF
{\True Tổng số gạo ban đầu là $360\text{ kg}$.}
{\True Số gạo đã bán là $120\text{ kg}$.}
{Số gạo còn lại là $200\text{ kg}$.}
{\True Số gạo còn lại là $240\text{ kg}$.}
\loigiai{
\begin{itemize}
    \item Tổng: $120 \times 3 = 360\text{ kg}$. Đúng.
    \item Đã bán: $360 : 3 = 120\text{ kg}$. Đúng.
    \item Còn lại: $360 - 120 = 240\text{ kg} \neq 200\text{ kg}$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}
