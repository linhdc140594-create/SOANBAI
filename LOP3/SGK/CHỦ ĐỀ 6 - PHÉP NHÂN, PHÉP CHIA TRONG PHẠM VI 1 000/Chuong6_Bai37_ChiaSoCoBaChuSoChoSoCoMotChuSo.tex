\begin{ex} %[3D6C37]

\choice
{$380\text{ kg}$}
{$420\text{ kg}$}
{\True $400\text{ kg}$}
{$450\text{ kg}$}
\loigiai{
Lần đầu: $650 : 5 = 130\text{ kg}$.\\
Đã xuất: $130 + 120 = 250\text{ kg}$.\\
Còn lại: $650 - 250 = 400\text{ kg}$.\\
\textbf{Đáp án của câu hỏi là C}
}
\end{ex}

\textbf{Phần 2. Trắc nghiệm đúng sai}
\setcounterpageref{ex}{0}

\begin{ex} %[3D6H37]
Xét tính đúng sai của các phép chia:
\choiceTF
{\True $484 : 4 = 121$.}
{\True $639 : 3 = 213$.}
{$525 : 5 = 15$.}
{\True $525 : 5 = 105$.}
\loigiai{
\begin{itemize}
    \item $484 : 4 = 121$: Đúng.
    \item $639 : 3 = 213$: Đúng.
    \item $525 : 5 = 105 \neq 15$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\begin{ex} %[3D6V37]
Một trang trại có 848 quả trứng gà xếp vào các vỉ, mỗi vỉ 8 quả. Xét tính đúng sai:
\choiceTF
{\True Số vỉ trứng xếp được là 106 vỉ.}
{\True Không còn thừa quả trứng nào (chia hết).}
{Nếu mỗi vỉ xếp 4 quả thì xếp được 210 vỉ.}
{\True Nếu mỗi vỉ xếp 4 quả thì xếp được 212 vỉ ($848 : 4 = 212$).}
\loigiai{
\begin{itemize}
    \item $848 : 8 = 106$ vỉ: Đúng.
    \item Chia hết: Đúng.
    \item $848 : 4 = 212$ vỉ $\neq 210$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

Tính thương: $693 : 3$.
\shortans{231}
\loigiai{
\textbf{Đáp số:} $231$\\
$693 : 3 = 231$.
}
