{$90000$}

\textbf{Đáp án của câu hỏi là B}

\textbf{Phần 2. Trắc nghiệm đúng sai}
\setcounterpageref{ex}{0}

\begin{ex} %[3D12H65]
Xét tính đúng sai của các phép toán:
\choiceTF
{\True $50000 + 50000 = 100000$.}
{\True $100000 - 40000 = 60000$.}
{$45000 + 35000 - 20000 = 50000$.}
{\True $45000 + 35000 - 20000 = 60000$.}
\loigiai{
\begin{itemize}
    \item $50000 + 50000 = 100000$: Đúng.
    \item $100000 - 40000 = 60000$: Đúng.
    \item $80000 - 20000 = 60000 \neq 50000$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\begin{ex} %[3D12V65]
Một quỹ khuyến học có $80\,000\,000$ đồng (hoặc $80\,000$ nghìn đồng). Đợt 1 trao $35\,000$, đợt 2 trao $25\,000$. Xét tính đúng sai:
\choiceTF
{\True Cả 2 đợt đã trao $60\,000$ (nghìn đồng).}
{\True Quỹ còn lại $20\,000$ (nghìn đồng).}
{Đợt 1 trao ít hơn đợt 2 là $10\,000$.}
{\True Đợt 1 trao nhiều hơn đợt 2 là $10\,000$ ($35000 - 25000 = 10000$).}
\loigiai{
\begin{itemize}
    \item Đã trao: $35000 + 25000 = 60000$. Đúng.
    \item Còn lại: $80000 - 60000 = 20000$. Đúng.
    \item Đợt 1 nhiều hơn đợt 2. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\textbf{Phần 3. Trả lời ngắn}
\setcounterpageref{ex}{0}

\begin{ex} %[3D12H65]
Tính tổng: $48\,250 + 31\,750$.
\shortans{80000}
\loigiai{
\textbf{Đáp số:} $80000$\\
$48250 + 31750 = 80000$.
}
\end{ex}

\begin{ex} %[3D12H65]
Tính hiệu: $90\,000 - 45\,500$.
\shortans{44500}
\loigiai{
\textbf{Đáp số:} $44500$\\
$90000 - 45500 = 44500$.
}
\end{ex}

\begin{ex} %[3D12V65]
Tính: $100\,000 - 38\,000 - 22\,000$.
\shortans{40000}
\loigiai{
\textbf{Đáp số:} $40000$\\
$100000 - 60000 = 40000$.
}
\end{ex}

\begin{ex} %[3D12V65]
Tìm $x$ biết: $x - 36\,000 = 44\,000$.
\shortans{80000}
\loigiai{
\textbf{Đáp số:} $80000$\\
$x = 44000 + 36000 = 80000$.
}
\end{ex}

\begin{ex} %[3D12V65]
Một kho thóc có $65\,000\text{ kg}$ thóc. Sau khi nhập thêm $15\,000\text{ kg}$ và xuất đi $30\,000\text{ kg}$ thì trong kho còn bao nhiêu ki-lô-gam thóc?
\shortans{50000}
\loigiai{
\textbf{Đáp số:} $50000$\\
$65000 + 15000 - 30000 = 50000\text{ kg}$.
}
\end{ex}

\begin{ex} %[3D12C65]
Tính nhanh: $45\,000 + 28\,000 + 15\,000 - 18\,000$.
\shortans{70000}
\loigiai{
\textbf{Đáp số:} $70000$\\
$(45000 + 15000) + (28000 - 18000) = 60000 + 10000 = 70000$.
}
\end{ex}
