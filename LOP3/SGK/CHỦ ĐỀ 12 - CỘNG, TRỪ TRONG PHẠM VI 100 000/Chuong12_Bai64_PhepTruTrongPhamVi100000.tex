\textbf{Phần 2. Trắc nghiệm đúng sai}
\setcounterpageref{ex}{0}

\begin{ex} %[3D12H64]
Xét tính đúng sai của các phép trừ:
\choiceTF
{\True $80000 - 25000 = 55000$.}
{\True $95600 - 42300 = 53300$.}
{$100000 - 65000 = 45000$.}
{\True $100000 - 65000 = 35000$.}
\loigiai{
\begin{itemize}
    \item $80000 - 25000 = 55000$: Đúng.
    \item $95600 - 42300 = 53300$: Đúng.
    \item $100000 - 65000 = 35000 \neq 45000$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\begin{ex} %[3D12V64]
Một kho hàng có $90\,000\text{ kg}$ xi măng. Lần 1 xuất $32\,000\text{ kg}$, lần 2 xuất $28\,000\text{ kg}$. Xét tính đúng sai:
\choiceTF
{\True Cả 2 lần xuất được $60\,000\text{ kg}$ xi măng.}
{\True Kho còn lại $30\,000\text{ kg}$ xi măng.}
{Lần 1 xuất ít hơn lần 2 là $4\,000\text{ kg}$.}
{\True Lần 1 xuất nhiều hơn lần 2 là $4\,000\text{ kg}$ ($32000 - 28000 = 4000$).}
\loigiai{
\begin{itemize}
    \item Đã xuất: $32000 + 28000 = 60000\text{ kg}$. Đúng.
    \item Còn lại: $90000 - 60000 = 30000\text{ kg}$. Đúng.
    \item Lần 1 nhiều hơn lần 2. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\textbf{Phần 3. Trả lời ngắn}
\setcounterpageref{ex}{0}

\begin{ex} %[3D12H64]
Tính hiệu: $87\,650 - 34\,210$.
\shortans{53440}
\loigiai{
\textbf{Đáp số:} $53440$\\
$87650 - 34210 = 53440$.
}
\end{ex}

\begin{ex} %[3D12H64]
Tính hiệu: $94\,500 - 46\,800$.
\shortans{47700}
\loigiai{
\textbf{Đáp số:} $47700$\\
$94500 - 46800 = 47700$.
}
\end{ex}

\begin{ex} %[3D12V64]
Tính: $100\,000 - 72\,500$.
\shortans{27500}
\loigiai{
\textbf{Đáp số:} $27500$\\

}
\end{ex}

\begin{ex} %[3D12V64]

\shortans{53000}
\loigiai{
\textbf{Đáp số:} $53000$\\

}
\end{ex}

\begin{ex} %[3D12V64]
Một bể chứa $60\,000\text{ l}$ nước, sau khi tháo bớt $24\,500\text{ l}$ thì bể còn lại bao nhiêu lít nước?
\shortans{35500}
\loigiai{
\textbf{Đáp số:} $35500$\\
$60000 - 24500 = 35500\text{ l}$.
}
\end{ex}

\begin{ex} %[3D12C64]
Tính nhanh: $95\,600 - 34\,000 - 15\,600$.
\shortans{46000}
\loigiai{
\textbf{Đáp số:} $46000$\\
$(95600 - 15600) - 34000 = 80000 - 34000 = 46000$.
}
\end{ex}
