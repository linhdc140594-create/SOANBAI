\textbf{Phần 2. Trắc nghiệm đúng sai}
\setcounterpageref{ex}{0}

\begin{ex} %[3H9H51]
Xét tính đúng sai về diện tích:
\choiceTF
{\True Diện tích hình $P$ nằm hoàn toàn trong hình $Q$ thì diện tích hình $P$ bé hơn diện tích hình $Q$.}
{\True Hai hình có cùng số ô vuông $1\text{ cm}^2$ thì có diện tích bằng nhau.}
{Xăng-ti-mét vuông là đơn vị đo độ dài.}
{\True Xăng-ti-mét vuông là đơn vị đo diện tích.}
\loigiai{
\begin{itemize}
    \item $P$ nằm trong $Q \Rightarrow S_P < S_Q$: Đúng.
    \item Cùng số ô vuông thì diện tích bằng nhau: Đúng.
    \item $\text{cm}^2$ là đơn vị đo diện tích (không phải độ dài). Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\begin{ex} %[3H9V51]
Hình $A$ có diện tích $48\text{ cm}^2$, hình $B$ có diện tích bằng $\dfrac{1}{4}$ hình $A$. Xét tính đúng sai:
\choiceTF
{\True Diện tích hình $B$ là $12\text{ cm}^2$.}
{\True Tổng diện tích hình $A$ và hình $B$ là $60\text{ cm}^2$.}
{Hình $A$ có diện tích lớn hơn hình $B$ là $30\text{ cm}^2$.}
{\True Hình $A$ có diện tích lớn hơn hình $B$ là $36\text{ cm}^2$ ($48 - 12 = 36$).}
\loigiai{
\begin{itemize}
    \item Hình $B$: $48 : 4 = 12\text{ cm}^2$. Đúng.
    \item Tổng: $48 + 12 = 60\text{ cm}^2$. Đúng.
    \item Hiệu: $48 - 12 = 36\text{ cm}^2 \neq 30\text{ cm}^2$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\textbf{Phần 3. Trả lời ngắn}
\setcounterpageref{ex}{0}

\begin{ex} %[3H9H51]
Tính: $65\text{ cm}^2 + 35\text{ cm}^2$.
\shortans{100}
\loigiai{
\textbf{Đáp số:} $100$\\
$65 + 35 = 100\text{ cm}^2$.
}
\end{ex}

\begin{ex} %[3H9H51]

$120 - 45 = 75\text{ cm}^2$.
}
\end{ex}
