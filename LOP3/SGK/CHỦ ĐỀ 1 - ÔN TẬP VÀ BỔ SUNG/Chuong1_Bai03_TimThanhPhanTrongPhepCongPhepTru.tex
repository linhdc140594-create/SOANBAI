\textbf{Phần 1. Trắc nghiệm lựa chọn}
\setcounterpageref{ex}{0}

\begin{ex} %[3D1N03]
Trong phép cộng $a + b = c$, $c$ được gọi là
\choice
{\True Tổng}
{Số hạng}
{Số bị trừ}
{Hiệu}
\loigiai{
Trong phép cộng, kết quả $c$ gọi là Tổng.\\
\textbf{Đáp án của câu hỏi là A}
}
\end{ex}

\begin{ex} %[3D1N03]
Trong phép trừ $x - y = z$, $x$ được gọi là
\choice
{Số trừ}
{\True Số bị trừ}
{Hiệu}
{Tổng}
\loigiai{
Trong phép trừ, $x$ gọi là Số bị trừ.\\
\textbf{Đáp án của câu hỏi là B}
}
\end{ex}

\begin{ex} %[3D1H03]
Muốn tìm số hạng chưa biết, ta lấy

{Số hạng đã biết trừ đi tổng}
{Số bị trừ cộng với hiệu}
{\True Tổng trừ đi số hạng đã biết}
\loigiai{
Muốn tìm số hạng chưa biết: Tổng trừ đi số hạng đã biết.\\
\textbf{Đáp án của câu hỏi là D}
}
\end{ex}

\textbf{Phần 2. Trắc nghiệm đúng sai}
\setcounterpageref{ex}{0}

\begin{ex} %[3D1H03]
Xét tính đúng sai của các quy tắc tìm thành phần phép tính:
\choiceTF
{\True Số hạng = Tổng $-$ Số hạng đã biết.}
{\True Số bị trừ = Hiệu $+$ Số trừ.}
{Số trừ = Số bị trừ $+$ Hiệu.}
{\True Số trừ = Số bị trừ $-$ Hiệu.}
\loigiai{
\begin{itemize}
    \item Tìm số hạng: Đúng.
    \item Tìm số bị trừ: Đúng.
    \item Số trừ $=$ Số bị trừ $-$ Hiệu (không phải cộng). Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\begin{ex} %[3D1V03]
Xét tính đúng sai khi tìm giá trị của $x$:
\choiceTF
{\True Nếu $x + 150 = 400$ thì $x = 250$.}
{\True Nếu $x - 70 = 180$ thì $x = 250$.}
{Nếu $300 - x = 120$ thì $x = 420$.}
{\True Nếu $300 - x = 120$ thì $x = 180$.}
\loigiai{
\begin{itemize}
    \item $x = 400 - 150 = 250$. Đúng.
    \item $x = 180 + 70 = 250$. Đúng.
    \item $x = 300 - 120 = 180 \neq 420$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\textbf{Phần 3. Trả lời ngắn}
\setcounterpageref{ex}{0}

\begin{ex} %[3D1H03]
Tìm $x$ biết: $x + 140 = 500$.
\shortans{360}
\loigiai{
\textbf{Đáp số:} $360$\\
$x = 500 - 140 = 360$.
}
\end{ex}

\begin{ex} %[3D1H03]
Tìm $x$ biết: $x - 220 = 380$.
\shortans{600}
\loigiai{
\textbf{Đáp số:} $600$\\
$x = 380 + 220 = 600$.
}
\end{ex}

\begin{ex} %[3D1V03]
Tìm $y$ biết: $750 - y = 320$.
\shortans{430}
\loigiai{
\textbf{Đáp số:} $430$\\
$y = 750 - 320 = 430$.
}
\end{ex}

\begin{ex} %[3D1V03]
Tìm $x$ biết: $x + 65 = 200 + 45$.
\shortans{180}
\loigiai{
\textbf{Đáp số:} $180$\\
$x + 65 = 245 \Rightarrow x = 245 - 65 = 180$.
}
\end{ex}

\begin{ex} %[3D1V03]
Một người có một túi bi, sau khi cho bạn 25 viên bi thì còn lại 45 viên bi. Hỏi lúc đầu túi có bao nhiêu viên bi?
\shortans{70}
\loigiai{
\textbf{Đáp số:} $70$\\
$45 + 25 = 70$ viên bi.
}
\end{ex}

\begin{ex} %[3D1C03]
Tìm $x$ biết: $500 - x - 40 = 260$.
\shortans{200}
\loigiai{
\textbf{Đáp số:} $200$\\
$500 - x = 260 + 40 = 300 \Rightarrow x = 500 - 300 = 200$.
}
\end{ex}
