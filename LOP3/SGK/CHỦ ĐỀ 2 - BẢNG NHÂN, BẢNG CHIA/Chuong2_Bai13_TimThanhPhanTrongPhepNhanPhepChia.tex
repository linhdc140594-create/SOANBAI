\textbf{Phần 1. Trắc nghiệm lựa chọn}
\setcounterpageref{ex}{0}

\begin{ex} %[3D2N13]
Trong phép tính $a \times b = c$, $a$ và $b$ được gọi là
\choice
{Số hạng}
{\True Thừa số}
{Số bị chia}
{Thương}
\loigiai{
Trong phép nhân, $a$ và $b$ gọi là Thừa số.\\
\textbf{Đáp án của câu hỏi là B}
}
\end{ex}

\begin{ex} %[3D2N13]
Trong phép chia $m : n = p$, $m$ được gọi là
\choice
{Số chia}
{Thương}
{\True Số bị chia}
{Thừa số}
\loigiai{
$m$ đứng trước dấu chia gọi là Số bị chia.\\
\textbf{Đáp án của câu hỏi là C}
}
\end{ex}

\begin{ex} %[3D2H13]
Muốn tìm thừa số chưa biết, ta lấy
\choice
{\True Tích chia cho thừa số đã biết}
{Tích nhân với thừa số đã biết}
{Tích cộng với thừa số đã biết}
{Tích trừ đi thừa số đã biết}
\loigiai{
Thừa số = Tích : Thừa số đã biết.\\
\textbf{Đáp án của câu hỏi là A}
}
\end{ex}

\begin{ex} %[3D2H13]
Tìm $x$ biết: $x \times 6 = 42$.
\choice
{$6$}
{$8$}
{\True $7$}
{$9$}
\loigiai{
$x = 42 : 6 = 7$.\\
\textbf{Đáp án của câu hỏi là C}
}
\end{ex}

\begin{ex} %[3D2H13]
Muốn tìm số bị chia, ta lấy
\choice
{Thương chia cho số chia}
{\True Thương nhân với số chia}
{Số chia chia cho thương}
{Thương cộng với số chia}
\loigiai{
Số bị chia = Thương $\times$ Số chia.\\
\textbf{Đáp án của câu hỏi là B}
}
\end{ex}

\begin{ex} %[3D2H13]

\begin{ex} %[3D2V13]

\textbf{Phần 2. Trắc nghiệm đúng sai}
\setcounterpageref{ex}{0}

\begin{ex} %[3D2H13]
Xét tính đúng sai của các quy tắc tìm thành phần phép tính:
\choiceTF
{\True Thừa số = Tích : Thừa số đã biết.}
{\True Số bị chia = Thương $\times$ Số chia.}
{Số chia = Thương $\times$ Số bị chia.}
{\True Số chia = Số bị chia : Thương.}
\loigiai{
\begin{itemize}
    \item Thừa số = Tích : Thừa số đã biết: Đúng.
    \item Số bị chia = Thương $\times$ Số chia: Đúng.
    \item Số chia = Số bị chia : Thương (không phải nhân). Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\begin{ex} %[3D2V13]
Xét tính đúng sai khi tìm giá trị $x$:
\choiceTF
{\True Nếu $x \times 9 = 54$ thì $x = 6$.}
{\True Nếu $x : 6 = 8$ thì $x = 48$.}
{Nếu $45 : x = 5$ thì $x = 225$.}
{\True Nếu $45 : x = 5$ thì $x = 9$.}
\loigiai{
\begin{itemize}
    \item $x = 54 : 9 = 6$. Đúng.
    \item $x = 8 \times 6 = 48$. Đúng.
    \item $x = 45 : 5 = 9 \neq 225$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\textbf{Phần 3. Trả lời ngắn}
\setcounterpageref{ex}{0}

\begin{ex} %[3D2H13]
Tìm $x$ biết: $x \times 7 = 35$.
\shortans{5}
\loigiai{
\textbf{Đáp số:} $5$\\
$x = 35 : 7 = 5$.
}
\end{ex}

\begin{ex} %[3D2H13]
Tìm $x$ biết: $x : 8 = 4$.
\shortans{32}
\loigiai{
\textbf{Đáp số:} $32$\\
$x = 4 \times 8 = 32$.
}
\end{ex}

\begin{ex} %[3D2V13]
Tìm $y$ biết: $63 : y = 9$.
\shortans{7}
\loigiai{
\textbf{Đáp số:} $7$\\
$y = 63 : 9 = 7$.
}
\end{ex}

\begin{ex} %[3D2V13]
Tìm $x$ biết: $x \times 6 = 20 + 28$.
\shortans{8}
\loigiai{
\textbf{Đáp số:} $8$\\
$x \times 6 = 48 \Rightarrow x = 48 : 6 = 8$.
}
\end{ex}

\begin{ex} %[3D2V13]
Tìm $x$ biết: $x : 5 = 12 - 4$.
\shortans{40}
\loigiai{
\textbf{Đáp số:} $40$\\

}
\end{ex}

\begin{ex} %[3D2C13]

\loigiai{

}
\end{ex}
