\textbf{Phần 2. Trắc nghiệm đúng sai}
\setcounterpageref{ex}{0}

\begin{ex} %[3D4H26]
Xét tính đúng sai của các phép chia:
\choiceTF
{\True $68 : 2 = 34$.}
{\True $81 : 3 = 27$.}
{$92 : 4 = 24$.}
{\True $92 : 4 = 23$.}
\loigiai{
\begin{itemize}
    \item $68 : 2 = 34$: Đúng.
    \item $81 : 3 = 27$: Đúng.
    \item $92 : 4 = 23 \neq 24$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\begin{ex} %[3D4V26]
Một trường tiểu học có 96 chiếc ghế chia đều cho 8 phòng học. Xét tính đúng sai:
\choiceTF
{\True Mỗi phòng học nhận được 12 chiếc ghế.}
{\True 3 phòng học nhận được 36 chiếc ghế.}
{5 phòng học nhận được 50 chiếc ghế.}
{\True 5 phòng học nhận được 60 chiếc ghế.}
\loigiai{
\begin{itemize}
    \item Mỗi phòng: $96 : 8 = 12$ ghế. Đúng.
    \item 3 phòng: $12 \times 3 = 36$ ghế. Đúng.
    \item 5 phòng: $12 \times 5 = 60$ ghế $\neq 50$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\textbf{Phần 3. Trả lời ngắn}
\setcounterpageref{ex}{0}

\begin{ex} %[3D4H26]
Tính thương: $56 : 4$.
\shortans{14}
\loigiai{
\textbf{Đáp số:} $14$\\
$56 : 4 = 14$.
}
\end{ex}

\begin{ex} %[3D4H26]
Tính thương: $78 : 3$.
\shortans{26}
\loigiai{
\textbf{Đáp số:} $26$\\
$78 : 3 = 26$.
}
\end{ex}

\begin{ex} %[3D4V26]

}
\end{ex}

\begin{ex} %[3D4V26]

$91 : 7 = 13\text{ cm}$.

\begin{ex} %[3D4V26]

\shortans{14}
\loigiai{

}
\end{ex}

\begin{ex} %[3D4C26]
Một hình chữ nhật có chu vi $64\text{ cm}$, chiều dài là $20\text{ cm}$. Tính chiều rộng của hình chữ nhật theo xăng-ti-mét.
\shortans{12}
\loigiai{

Nửa chu vi: $64 : 2 = 32\text{ cm}$. Chiều rộng: $32 - 20 = 12\text{ cm}$.
}
\end{ex}
