\textbf{Phần 1. Trắc nghiệm lựa chọn}
\setcounterpageref{ex}{0}

\begin{ex} %[3D4N27]
Muốn giảm một số đi nhiều lần, ta lấy
\choice
{Số đó trừ đi số lần}
{Số đó nhân với số lần}
{\True Số đó chia cho số lần}
{Số đó cộng với số lần}
\loigiai{
Muốn giảm một số đi nhiều lần, ta lấy số đó chia cho số lần.\\
\textbf{Đáp án của câu hỏi là C}
}
\end{ex}

\begin{ex} %[3D4N27]

$24 : 4 = 6$.\\
\textbf{Đáp án của câu hỏi là A}

\textbf{Phần 2. Trắc nghiệm đúng sai}
\setcounterpageref{ex}{0}

\begin{ex} %[3D4H27]
Xét tính đúng sai về giảm một số đi nhiều lần:
\choiceTF
{\True Giảm 32 đi 4 lần được 8.}
{\True Giảm 48 đi 6 lần được 8.}
{Giảm 40 đi 5 lần được 35.}
{\True Giảm 40 đi 5 lần được 8.}
\loigiai{
\begin{itemize}
    \item $32 : 4 = 8$: Đúng.
    \item $48 : 6 = 8$: Đúng.
    \item Giảm đi số lần là phép chia ($40 : 5 = 8 \neq 35$). Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\begin{ex} %[3D4V27]
Đoạn thẳng $AB = 60\text{ cm}$, đoạn thẳng $CD$ giảm 5 lần so với $AB$. Xét tính đúng sai:
\choiceTF
{\True Đoạn thẳng $CD$ dài $12\text{ cm}$.}
{\True Tổng độ dài hai đoạn thẳng là $72\text{ cm}$.}
{Đoạn thẳng $AB$ dài hơn đoạn thẳng $CD$ là $50\text{ cm}$.}
{\True Đoạn thẳng $AB$ dài hơn đoạn thẳng $CD$ là $48\text{ cm}$.}
\loigiai{
\begin{itemize}
    \item $CD = 60 : 5 = 12\text{ cm}$. Đúng.
    \item Tổng: $60 + 12 = 72\text{ cm}$. Đúng.
    \item Hiệu: $60 - 12 = 48\text{ cm} \neq 50\text{ cm}$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\shortans{9}
\loigiai{

$54 : 6 = 9$.
}
\end{ex}

\begin{ex} %[3D4C27]
Tìm một số biết rằng nếu giảm số đó đi 3 lần rồi cộng thêm 18 thì được 30.
\shortans{36}
\loigiai{
\textbf{Đáp số:} $36$\\
Số đó $: 3 = 30 - 18 = 12 \Rightarrow$ Số đó là $12 \times 3 = 36$.
}
