\textbf{Phần 2. Trắc nghiệm đúng sai}
\setcounterpageref{ex}{0}

\begin{ex} %[3D8H47]
Xét tính đúng sai của các chữ số La Mã:
\choiceTF
{\True $\text{III}$ biểu thị số 3.}
{\True $\text{VII}$ biểu thị số 7.}
{$\text{XII}$ biểu thị số 20.}
{\True $\text{XII}$ biểu thị số 12.}
\loigiai{
\begin{itemize}
    \item $\text{III} = 3$: Đúng.
    \item $\text{VII} = 7$: Đúng.
    \item $\text{XII} = 12 \neq 20$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\begin{ex} %[3D8V47]
Xét thứ tự các số La Mã:
\choiceTF
{\True $\text{IV} < \text{V} < \text{VI}$.}
{\True $\text{IX} < \text{X} < \text{XI}$.}
{$\text{XIV} > \text{XVI}$.}
{\True $\text{XIV} < \text{XVI}$ (vì $14 < 16$).}
\loigiai{
\begin{itemize}
    \item $4 < 5 < 6$: Đúng.
    \item $9 < 10 < 11$: Đúng.
    \item $14 < 16$ (không lớn hơn). Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}
