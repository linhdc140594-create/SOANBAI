\begin{ex} %[3D14C71]
Một kho hàng có $96\,000\text{ kg}$ thóc. Lần đầu xuất đi $\dfrac{1}{4}$ số thóc, lần sau xuất đi $\dfrac{1}{3}$ số thóc CÒN LẠI. Số thóc còn lại sau hai lần xuất là
\choice
{$46000\text{ kg}$}
{$50000\text{ kg}$}
{\True $48000\text{ kg}$}
{$52000\text{ kg}$}
\loigiai{
Lần 1: $96000 : 4 = 24000\text{ kg}$.\\
Còn lại sau lần 1: $96000 - 24000 = 72000\text{ kg}$.\\
Lần 2: $72000 : 3 = 24000\text{ kg}$.\\
Còn lại cuối cùng: $72000 - 24000 = 48000\text{ kg}$.\\
\textbf{Đáp án của câu hỏi là C}
}
\end{ex}

\textbf{Phần 2. Trắc nghiệm đúng sai}
\setcounterpageref{ex}{0}

\begin{ex} %[3D14H71]
Xét tính đúng sai của các phép chia:
\choiceTF
{\True $64820 : 2 = 32410$.}
{\True $93690 : 3 = 31230$.}
{$55500 : 5 = 1100$.}
{\True $55500 : 5 = 11100$.}
\loigiai{
\begin{itemize}
    \item $64820 : 2 = 32410$: Đúng.
    \item $93690 : 3 = 31230$: Đúng.
    \item $55500 : 5 = 11100 \neq 1100$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}

\begin{ex} %[3D14V71]
Một phân xưởng may được $48\,800$ chiếc khẩu trang đóng đều vào 4 thùng. Xét tính đúng sai:
\choiceTF
{\True Mỗi thùng có $12\,200$ chiếc khẩu trang.}
{\True 2 thùng có $24\,400$ chiếc khẩu trang.}
{3 thùng có $36\,500$ chiếc khẩu trang.}
{\True 3 thùng có $36\,600$ chiếc khẩu trang ($12200 \times 3 = 36600$).}
\loigiai{
\begin{itemize}
    \item Mỗi thùng: $48800 : 4 = 12200$ chiếc. Đúng.
    \item 2 thùng: $12200 \times 2 = 24400$ chiếc. Đúng.
    \item 3 thùng: $12200 \times 3 = 36600 \neq 36500$. Ý 3 Sai, Ý 4 Đúng.
\end{itemize}
\begin{itemize}
    \item a) \textbf{Đúng}.
    \item b) \textbf{Đúng}.
    \item c) \textbf{Sai}.
    \item d) \textbf{Đúng}.
\end{itemize}
}
\end{ex}
