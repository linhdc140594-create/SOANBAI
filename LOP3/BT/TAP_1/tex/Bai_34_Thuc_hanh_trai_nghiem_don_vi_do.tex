\begin{ex}
Đoạn que dài 15 cm. Bạn Nam đo được bao nhiêu mi-li-mét?
\shortans{150}
\loigiai{
$15\text{ cm} = 150\text{ mm}$.
}
\end{ex}

\begin{ex}
Một quả chuối nặng khoảng 120 g. Hỏi 3 quả chuối như thế nặng khoảng bao nhiêu gam?
\shortans{360}
\loigiai{
$120 \times 3 = 360$ g.
}
\end{ex}

\begin{ex}

\end{ex}
