\begin{ex}

\loigiai{
An có tất cả: $4 + 3 = 7$ quả bóng bay.
}

\begin{ex}

\shortans{9}
\loigiai{
Số chim trên cây là: $5 + 4 = 9$ con.
}
\end{ex}
