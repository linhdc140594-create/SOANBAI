\begin{ex}
Tính: $3 + 4 + 3$.
\shortans{10}
\loigiai{
Thực hiện: $3 + 4 = 7$; $7 + 3 = 10$.
\textbf{Đáp số:} $10$.
}
\end{ex}

\textbf{Đáp số:} $3$.
}
\end{ex}

\begin{ex}
Tìm $x$ biết: $x + 2 = 5 + 4$.
\shortans{7}
\loigiai{
Vế phải: $5 + 4 = 9$. Ta có: $x + 2 = 9 \Rightarrow x = 9 - 2 = 7$.
\textbf{Đáp số:} $7$.
}
\end{ex}

\begin{ex}

\shortans{4}
\loigiai{
Vế phải: $8 - 2 = 6$. Ta có: $10 - y = 6 \Rightarrow y = 10 - 6 = 4$.
\textbf{Đáp số:} $4$.
}
\end{ex}

\shortans{9}
