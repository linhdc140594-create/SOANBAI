% --- KHỐI 1: CÔNG THỨC SXQ VÀ STP ---
\noindent
\begin{tcolorbox}[
  enhanced,
  title={\footnotesize\bfseries\color{white}\faCalculator\ 1. Công thức Diện tích Xung quanh và Toàn phần},
  colback=blue!2!white,
  colframe=blue!75!black,
  arc=2.5mm, boxrule=0.8pt, top=1.8mm, bottom=1.8mm, left=2.5mm, right=2.5mm
]
  \footnotesize
  Cho hình hộp chữ nhật có chiều dài $a$, chiều rộng $b$, chiều cao $c$ (hoặc $h$):\\
  $\bullet$ \textbf{Diện tích xung quanh ($S_{xq}$):} Bằng chu vi mặt đáy nhân với chiều cao:\\
  \begin{tcolorbox}[colback=white, colframe=blue!75!black, boxrule=0.6pt, arc=1.5mm, halign=center, top=1.2mm, bottom=1.2mm]
    {\scriptsize\bfseries\color{blue!90!black}$S_{xq} = (a + b) \times 2 \times c = P_{\text{đáy}} \times c$}
  \end{tcolorbox}
  $\bullet$ \textbf{Diện tích toàn phần ($S_{tp}$):} Bằng diện tích xung quanh cộng với diện tích hai mặt đáy:\\
  \begin{tcolorbox}[colback=white, colframe=purple!70!black, boxrule=0.6pt, arc=1.5mm, halign=center, top=1.2mm, bottom=1.2mm]
    {\scriptsize\bfseries\color{purple!90!black}$S_{tp} = S_{xq} + S_{\text{đáy}} \times 2 = S_{xq} + (a \times b) \times 2$}
  \end{tcolorbox}
\end{tcolorbox}
