    \textit{VD:} $2\text{h } 45\text{p} + 3\text{h } 30\text{p} = 5\text{h } 75\text{p} = \mathbf{6\text{h } 15\text{p}}$.

    \footnotesize
    $\bullet$ \textbf{Đặt tính}: Thẳng cột theo từng loại đơn vị.\\
    $\bullet$ \textbf{Đổi mượn (NẾU CẦN)}: Nếu số đo ở đơn vị bé của số bị trừ nhỏ hơn số trừ, ta \textbf{mượn 1 đơn vị lớn đổi thành 60 đơn vị bé}.\\
    \textit{VD:} $5\text{ giờ } 15\text{ phút} - 2\text{ giờ } 40\text{ phút}$:\\
    Đổi: $5\text{h } 15\text{p} = 4\text{h } 75\text{p}$.\\
    Trừ: $4\text{h } 75\text{p} - 2\text{h } 40\text{p} = \mathbf{2\text{ giờ } 35\text{ phút}}$.
  \end{tcolorbox}

    $3\text{ ngày } 8\text{ giờ} - 1\text{ ngày } 14\text{ giờ}$:\\

    $\triangleright$ Trừ: $2\text{ ngày } 32\text{ giờ} - 1\text{ ngày } 14\text{ giờ} = \mathbf{1\text{ ngày } 18\text{ giờ}}$.
