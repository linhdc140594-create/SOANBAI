% --- KHỐI 1: BỐN BƯỚC GIẢI TOÁN KINH ĐIỂN ---
\noindent
\begin{tcolorbox}[
  enhanced,
  title={\footnotesize\bfseries\color{white}\faKey\ 1. Bốn bước vàng giải bài toán Tổng -- Tỉ},
  colback=blue!2!white,
  colframe=blue!75!black,
  arc=2.5mm, boxrule=0.8pt, top=1.8mm, bottom=1.8mm, left=2.5mm, right=2.5mm
]
  \footnotesize
  \textbf{Bước 1: Vẽ sơ đồ đoạn thẳng}\\
  $\bullet$ Dựa vào tỉ số $\dfrac{a}{b}$: Số bé vẽ $a$ phần bằng nhau, Số lớn vẽ $b$ phần như thế. Thể hiện Tổng ở móc nhọn.\\
  \textbf{Bước 2: Tìm tổng số phần bằng nhau}\\
  \begin{tcolorbox}[colback=white, colframe=blue!75!black, boxrule=0.5pt, arc=1.5mm, halign=center, top=1mm, bottom=1mm]
    {\scriptsize\bfseries\color{blue!90!black}$\text{Tổng số phần bằng nhau} = a + b\text{ (phần)}$}
  \end{tcolorbox}
  \textbf{Bước 3: Tìm giá trị của một phần:}\\
  $\bullet$ $\text{Giá trị } 1\text{ phần} = \text{Tổng hai số} : \text{Tổng số phần}$.\\
  \textbf{Bước 4: Tìm số bé và số lớn:}\\
  $\bullet$ $\text{Số bé} = \text{Giá trị } 1\text{ phần} \times a$ \quad | \quad $\text{Số lớn} = \text{Tổng} - \text{Số bé}$ (hoặc Giá trị $1$ phần $\times b$).
\end{tcolorbox}
