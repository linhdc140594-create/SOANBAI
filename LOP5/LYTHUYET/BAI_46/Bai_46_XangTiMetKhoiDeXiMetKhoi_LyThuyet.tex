% --- KHỐI 2: QUAN HỆ GẤP 1000 LẦN ---
\noindent
\begin{tcolorbox}[
  enhanced,
  title={\footnotesize\bfseries\color{white}\faExchange\ 3. Mối quan hệ giữa $\text{dm}^3$ và $\text{cm}^3$ (Gấp kém 1000 lần)},
  colback=purple!2!white,
  colframe=purple!70!black,
  arc=2.5mm, boxrule=0.8pt, top=1.8mm, bottom=1.8mm, left=2.5mm, right=2.5mm
]
  \footnotesize
  Một hình lập phương cạnh $1\text{ dm} = 10\text{ cm}$ được xếp bởi:\\
  $10 \text{ hàng} \times 10 \text{ khối} \times 10 \text{ lớp} = 1\,000\text{ khối lập phương cạnh } 1\text{ cm}$.\\
  \begin{tcolorbox}[colback=white, colframe=purple!70!black, boxrule=0.6pt, arc=1.5mm, halign=center, top=1.2mm, bottom=1.2mm]
    {\scriptsize\bfseries\color{purple!90!black}$1\text{ dm}^3 = 1\,000\text{ cm}^3$ \qquad | \qquad $1\text{ cm}^3 = \dfrac{1}{1000}\text{ dm}^3 = 0,001\text{ dm}^3$}
  \end{tcolorbox}
  $\triangleright$ Mỗi đơn vị thể tích liền nhau hơn kém nhau \textbf{1 000 lần} (ứng với \textbf{3 chữ số} mỗi cột!).
\end{tcolorbox}
