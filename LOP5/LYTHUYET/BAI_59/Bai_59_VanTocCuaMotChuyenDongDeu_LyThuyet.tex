\begin{tcolorbox}[
  enhanced,
  title={\footnotesize\bfseries\color{white}\faExchange\ 3. Quy tắc chuyển đổi giữa $\text{km/h}$ và $\text{m/s}$},
  colback=purple!2!white,
  colframe=purple!70!black,
  arc=2.5mm, boxrule=0.8pt, top=1.8mm, bottom=1.8mm, left=2.5mm, right=2.5mm
]
  \footnotesize
  $1\text{ km/giờ} = \dfrac{1\,000\text{ m}}{3\,600\text{ giây}} = \dfrac{1}{3,6}\text{ m/giây}$. Do đó:\\
  \begin{tcolorbox}[colback=white, colframe=purple!70!black, boxrule=0.6pt, arc=1.5mm, halign=center, top=1.2mm, bottom=1.2mm]
    {\scriptsize\bfseries\color{purple!90!black}$\text{Từ m/giây} \longrightarrow \text{km/giờ: \textbf{Nhân với 3,6}}$ \qquad | \qquad $\text{Từ km/giờ} \longrightarrow \text{m/giây: \textbf{Chia cho 3,6}}$}
  \end{tcolorbox}
  \textit{VD:} $10\text{ m/s} = 10 \times 3,6 = \mathbf{36\text{ km/h}}$ \quad | \quad $54\text{ km/h} = 54 : 3,6 = \mathbf{15\text{ m/s}}$.
\end{tcolorbox}
