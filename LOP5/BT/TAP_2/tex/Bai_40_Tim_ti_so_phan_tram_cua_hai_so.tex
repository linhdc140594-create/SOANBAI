% Câu 1
\begin{ex}
  Tìm tỉ số phần trăm của hai số 15 và 60.
  \shortans{25}
  \loigiai{
    Tỉ số phần trăm là: (15 : 60) x 100% = 25%.
  }
\end{ex}
