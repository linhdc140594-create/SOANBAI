% Câu 1
\begin{ex}
  Tung một đồng xu 20 lần, xuất hiện mặt ngửa 12 lần. Tỉ số phần trăm số lần xuất hiện mặt ngửa là bao nhiêu phần trăm?
  \shortans{60}
  \loigiai{
    $(12 : 20) \times 100\% = 60\%$.
  }
\end{ex}

% Câu 2
\begin{ex}
  Ở thực nghiệm trên, số lần xuất hiện mặt sấp là bao nhiêu lần?
  \shortans{8}
  \loigiai{
    $20 - 12 = 8$ lần.
  }
\end{ex}
