% Câu 1
\begin{ex}
  Một hình hộp chữ nhật có chiều dài $11\text{ cm}$, chiều rộng $6\text{ cm}$ và chiều cao $5\text{ cm}$. Diện tích xung quanh của nó là bao nhiêu xăng-ti-mét vuông?
  \shortans{170}
  \loigiai{
    Chu vi đáy: (11 + 6) x 2 = 34 cm. Diện tích xung quanh: 34 x 5 = 170 cm2.
  }
\end{ex}
