% Câu 1
\begin{ex}
  Đổi $2\text{ dm}^3$ thành bao nhiêu xăng-ti-mét khối?
  \shortans{2000}
  \loigiai{
    1 dm3 = 1000 cm3. Do đó 2 dm3 = 2000 cm3.
  }
\end{ex}
