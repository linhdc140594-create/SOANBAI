% Câu 1
\begin{ex}
  Một hình lập phương có cạnh $3\text{ cm}$. Diện tích xung quanh của nó là bao nhiêu xăng-ti-mét vuông?
  \shortans{36}
  \loigiai{
    Diện tích xung quanh là: 3 x 3 x 4 = 36 cm2.
  }
\end{ex}
