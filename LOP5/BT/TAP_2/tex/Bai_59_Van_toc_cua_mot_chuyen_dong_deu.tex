% Câu 1
\begin{ex}
  Một người đi xe máy đi được quãng đường $120\text{ km}$ trong $3$ giờ. Vận tốc của người đó là bao nhiêu km/h?
  \shortans{40}
  \loigiai{
    $120 : 3 = 40\text{ km/h}$.
  }
\end{ex}
