% Câu 1
\begin{ex}
  Một hình lập phương có cạnh $1\text{ cm}$. Thể tích của nó là bao nhiêu xăng-ti-mét khối?
  \shortans{1}
  \loigiai{
    Thể tích là: 1 x 1 x 1 = 1 cm3.
  }
\end{ex}
