% Câu 1
\begin{ex}
  Sử dụng máy tính cầm tay thực hiện phép tính: $135 + 27$.
  \shortans{162}
  \loigiai{
    Bấm phím [135] [+] [27] [=] ta được kết quả: 162.
  }
\end{ex}
