% Câu 11
\begin{ex}
  Một hộp sữa bột cân nặng $0{,}45\text{ kg}$. Một thùng có $20$ hộp sữa như thế. Cả thùng sữa cân nặng bao nhiêu ki-lô-gam (bỏ qua vỏ thùng)?
  \shortans{9}
  \loigiai{
    Khối lượng là: $0{,}45 \times 20 = 9\text{ kg}$.
  }
\end{ex}

% Câu 12
\begin{ex}
  Một tấm vải có giá $45\,000$ đồng mỗi mét. Mua $2{,}4\text{ m}$ vải hết bao nhiêu nghìn đồng?
  \shortans{108}
  \loigiai{
    $45 \times 2{,}4 = 108$ nghìn đồng.
  }
\end{ex}

% Câu 13
\begin{ex}

  \shortans{18}
  \loigiai{
    Quãng đường là: $4{,}5 \times 4 = 18\text{ km}$.
  }
\end{ex}

% Câu 14
\begin{ex}
  Một mảnh vườn hình vuông có cạnh dài $6{,}5\text{ m}$. Chu vi mảnh vườn đó là bao nhiêu mét?
  \shortans{26}
  \loigiai{
    Chu vi là: $6{,}5 \times 4 = 26\text{ m}$.
  }
\end{ex}

% Câu 15
\begin{ex}
  Một hình chữ nhật có chiều dài $8\text{ m}$, chiều rộng $2{,}5\text{ m}$. Diện tích hình chữ nhật là bao nhiêu mét vuông?
  \shortans{20}
  \loigiai{
    Diện tích là: $8 \times 2{,}5 = 20\text{ m}^2$.
  }
\end{ex}
