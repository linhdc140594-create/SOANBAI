\begin{ex}

  \shortans{475}
  \loigiai{
    $4{,}75 \times 100 = 475$.
  }
\end{ex}

% Câu 18
\begin{ex}
  Một bao gạo nặng $50\text{ kg}$. Hỏi $100$ bao gạo như thế nặng bao nhiêu tạ?
  \shortans{50}
  \loigiai{
    Tổng là $50 \times 100 = 5\,000\text{ kg} = 50\text{ tạ}$.
  }
\end{ex}

% Câu 19
\begin{ex}

  \shortans{50}
  \loigiai{
    $0{,}05 \times 1\,000 = 50\text{ mm}$.
  }
\end{ex}

% Câu 24
\begin{ex}
  Tìm $y$ biết: $y : 0{,}25 + y : 0{,}5 = 30$.
  \shortans{5}
  \loigiai{

  }
\end{ex}
