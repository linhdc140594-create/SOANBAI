% Câu 1
\begin{ex}
  Tính kết quả: $24{,}6 + 15{,}4$.
  \shortans{40}
  \loigiai{
    $24{,}6 + 15{,}4 = 40$.
  }
\end{ex}

% Câu 2
\begin{ex}
  Tính kết quả: $50 - 18{,}5$. Lấy kết quả nhân 2 được bao nhiêu?
  \shortans{63}
  \loigiai{
    $31{,}5 \times 2 = 63$.
  }
\end{ex}

% Câu 11
\begin{ex}
  Một hình chữ nhật có chiều dài $12\text{ m}$, chiều rộng $4{,}5\text{ m}$. Diện tích hình chữ nhật là bao nhiêu mét vuông?
  \shortans{54}
  \loigiai{
    $12 \times 4{,}5 = 54\text{ m}^2$.
  }
\end{ex}

% Câu 12
\begin{ex}
  Một ô tô trong $3\text{ giờ}$ đi được $135\text{ km}$. Hỏi trong $5\text{ giờ}$ ô tô đó đi được bao nhiêu ki-lô-mét?
  \shortans{225}
  \loigiai{
    Vận tốc: $135 : 3 = 45\text{ km/h}$. Trong 5 giờ: $45 \times 5 = 225\text{ km}$.
  }
\end{ex}

  \shortans{40}
  \loigiai{
    $100 - 10 \times 6 = 100 - 60 = 40$.
  }
\end{ex}

\end{ex}

% Câu 15
\begin{ex}
  Một người thợ làm $5$ sản phẩm hết $4\text{ giờ}$. Hỏi làm $15$ sản phẩm như thế hết bao nhiêu giờ?
  \shortans{12}
  \loigiai{
    $15$ sản phẩm gấp 5 sản phẩm là 3 lần. Thời gian: $4 \times 3 = 12$ giờ.
  }
\end{ex}

% Câu 16
\begin{ex}
  Có $36$ lít nước mắm chia đều vào các chai loại $0{,}75$ lít. Chia được bao nhiêu chai?
  \shortans{48}
  \loigiai{
    $36 : 0{,}75 = 48$ chai.
  }
\end{ex}

% Câu 18
\begin{ex}
  Một can chứa $20$ lít dầu, mỗi lít dầu cân nặng $0{,}8\text{ kg}$. Biết vỏ can nặng $1\text{ kg}$. Can dầu đó nặng tất cả bao nhiêu ki-lô-gam?
  \shortans{17}
  \loigiai{
    Dầu nặng $20 \times 0{,}8 = 16\text{ kg}$. Cả can nặng $16 + 1 = 17\text{ kg}$.
  }
\end{ex}

% Câu 19
\begin{ex}
  Tìm số tự nhiên $y$ biết: $y \times 0{,}5 + y \times 0{,}5 = 45$.
  \shortans{45}
  \loigiai{
    $y \times (0{,}5 + 0{,}5) = y \times 1 = 45 \implies y = 45$.
  }
\end{ex}

% Câu 21
\begin{ex}
  Một mảnh vườn hình chữ nhật có chu vi $50\text{ m}$, chiều rộng $10{,}5\text{ m}$. Chiều dài mảnh vườn là bao nhiêu phần mười mét? (Điền chiều dài nhân 10).
  \shortans{145}
  \loigiai{
    Nửa chu vi 25 m. Chiều dài là $25 - 10{,}5 = 14{,}5\text{ m}$. Nhân 10 được 145.
  }
\end{ex}

% Câu 22
\begin{ex}
  Một cửa hàng nhập về $2$ tạ đường, đã bán được $145{,}5\text{ kg}$. Cửa hàng còn lại bao nhiêu phần mười ki-lô-gam đường? (Điền số kg còn lại nhân 10).
  \shortans{545}
  \loigiai{
    $200 - 145{,}5 = 54{,}5\text{ kg}$. Nhân 10 được 545.
  }
\end{ex}

% Câu 24
\begin{ex}

    $50 \times 0{,}6 = 30\text{ kg}$.
  }
\end{ex}
