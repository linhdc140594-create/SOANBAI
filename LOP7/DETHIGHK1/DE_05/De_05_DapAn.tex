\begin{ex}
Tìm số tự nhiên $x$, biết: $2^{x+3} + 2^x = 144$. \shortans{$x = 4$}
\loigiai{
Ta có:
\[ 2^{x+3} + 2^x = 144 \Leftrightarrow 2^x \cdot 2^3 + 2^x = 144 \Leftrightarrow 2^x \cdot (8 + 1) = 144 \Leftrightarrow 2^x \cdot 9 = 144 \Leftrightarrow 2^x = 16 = 2^4. \]
Do đó $x = 4$.
Đáp án: $4$.
}
\end{ex}

\begin{ex}
Cho hai đường thẳng song song $a$ và $b$. Đường thẳng $c$ cắt $a$ và $b$ lần lượt tại hai điểm $A$ và $B$. Biết góc $\widehat{A}_1$ và góc $\widehat{B}_1$ là hai góc trong cùng phía, trong đó $\widehat{A}_1 = 65^\circ$. Tính số đo góc $\widehat{B}_1$. \shortans{$\widehat{B}_1 = 115^\circ$}
\loigiai{
Vì $a \parallel b$ nên hai góc trong cùng phía bù nhau:
\[ \widehat{A}_1 + \widehat{B}_1 = 180^\circ \Rightarrow 65^\circ + \widehat{B}_1 = 180^\circ \Rightarrow \widehat{B}_1 = 180^\circ - 65^\circ = 115^\circ. \]
Đáp án: $115^\circ$.
}
\end{ex}

\begin{enumerate}[label=\textbf{\alph*)}, leftmargin=0.8cm, itemsep=1pt]
\item $\dfrac{-5}{9} \cdot \dfrac{3}{11} + \dfrac{-5}{9} \cdot \dfrac{8}{11}$

&= 1 + 5 = 6.
\end{align*}

\textbf{c)} Ta có:
\[ \dfrac{1}{4} - 2 : \dfrac{4}{5} = \dfrac{1}{4} - 2 \cdot \dfrac{5}{4} = \dfrac{1}{4} - \dfrac{5}{2} = \dfrac{1 - 10}{4} = -\dfrac{9}{4}. \]

}
\end{ex}

Suy ra $\Delta AMB = \Delta AMC$ (c-g-c).

\textbf{b)} Vì $\Delta AMB = \Delta AMC$ (chứng minh trên) nên $MB = MC$ (hai cạnh tương ứng).
Mà điểm $M$ nằm giữa hai điểm $B$ và $C$ ($M \in BC$), suy ra $M$ là trung điểm của cạnh $BC$.

}
\end{ex}
\vspace{1.5mm}

\begin{ex}
Cho $A = \dfrac{1}{2^2} + \dfrac{1}{3^2} + \dfrac{1}{4^2} + \dots + \dfrac{1}{100^2}$. Chứng minh rằng $A < 1$.
\loigiai{
Ta có nhận xét với mọi số nguyên dương $n \ge 2$:
\[ \dfrac{1}{n^2} < \dfrac{1}{n(n-1)} = \dfrac{1}{n-1} - \dfrac{1}{n}. \]
Áp dụng với các số hạng của $A$:
\begin{align*}
A &< \left(1 - \dfrac{1}{2}\right) + \left(\dfrac{1}{2} - \dfrac{1}{3}\right) + \left(\dfrac{1}{3} - \dfrac{1}{4}\right) + \dots + \left(\dfrac{1}{99} - \dfrac{1}{100}\right) \\
&= 1 - \dfrac{1}{100} = \dfrac{99}{100}. \end{align*}
Vì $\dfrac{99}{100} < 1$ nên $A < 1$.
}
\end{ex}
